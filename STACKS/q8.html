Q8. Sort the stack in ascending order using integer array.

Program

#include <stdio.h>

int a[5], top = -1;

void push()
{
    int x;

    if(top == 4)
        printf("Stack is Full");
    else
    {
        printf("Enter number: ");
        scanf("%d", &x);
        top++;
        a[top] = x;
    }
}

void sort()
{
    int i, j, temp;

    for(i = 0; i < top; i++)
    {
        for(j = 0; j < top - i; j++)
        {
            if(a[j] > a[j + 1])
            {
                temp = a[j];
                a[j] = a[j + 1];
                a[j + 1] = temp;
            }
        }
    }

    printf("Stack in ascending order: ");

    for(i = 0; i <= top; i++)
        printf("%d ", a[i]);
}

int main()
{
    int n, i;

    printf("Enter number of elements (max 5): ");
    scanf("%d", &n);

    if(n < 1 || n > 5)
    {
        printf("Invalid size");
        return 0;
    }

    for(i = 0; i < n; i++)
        push();

    sort();

    return 0;
}
