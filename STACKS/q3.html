Q3. Create a stack implementing the operations on an integer nodes (dynamically) (push, pop, peek, isempty, isfull)

Program

#include <stdio.h>
#include <stdlib.h>

struct node
{
    int data;
    struct node *next;
};

struct node *top = NULL;

void push()
{
    struct node *p;
    int x;

    p = (struct node *)malloc(sizeof(struct node));

    if(p == NULL)
        printf("Stack is Full");
    else
    {
        printf("Enter number: ");
        scanf("%d", &x);

        p->data = x;
        p->next = top;
        top = p;
    }
}

void pop()
{
    struct node *p;

    if(top == NULL)
        printf("Stack is Empty");
    else
    {
        p = top;
        printf("Deleted: %d", top->data);
        top = top->next;
        free(p);
    }
}

void peek()
{
    if(top == NULL)
        printf("Stack is Empty");
    else
        printf("Top value: %d", top->data);
}

void isempty()
{
    if(top == NULL)
        printf("Stack is Empty");
    else
        printf("Stack is not Empty");
}

void isfull()
{
    struct node *p;

    p = (struct node *)malloc(sizeof(struct node));

    if(p == NULL)
        printf("Stack is Full");
    else
    {
        printf("Stack is not Full");
        free(p);
    }
}

int main()
{
    int ch;

    while(1)
    {
        printf("\n\n1. Push");
        printf("\n2. Pop");
        printf("\n3. Peek");
        printf("\n4. IsEmpty");
        printf("\n5. IsFull");
        printf("\n6. Exit");
        printf("\nEnter choice: ");
        scanf("%d", &ch);

        if(ch == 1)
            push();
        else if(ch == 2)
            pop();
        else if(ch == 3)
            peek();
        else if(ch == 4)
            isempty();
        else if(ch == 5)
            isfull();
        else if(ch == 6)
            break;
        else
            printf("Wrong choice");
    }

    return 0;
}
