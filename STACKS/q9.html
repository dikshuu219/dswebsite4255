Q9. Enter the infix expression from user and convert into prefix expression.

Program

#include <stdio.h>
#include <string.h>

char stack[100];
int top = -1;

void push(char x)
{
    stack[++top] = x;
}

char pop()
{
    return stack[top--];
}

int priority(char x)
{
    if(x == '*' || x == '/')
        return 2;
    if(x == '+' || x == '-')
        return 1;
    return 0;
}

int main()
{
    char infix[100], prefix[100], temp;
    int i, j = 0, n;

    printf("Enter infix expression: ");
    scanf("%s", infix);

    n = strlen(infix);

    for(i = 0; i < n/2; i++)
    {
        temp = infix[i];
        infix[i] = infix[n-i-1];
        infix[n-i-1] = temp;
    }

    for(i = 0; i < n; i++)
    {
        if(infix[i] == '(')
            infix[i] = ')';
        else if(infix[i] == ')')
            infix[i] = '(';
    }

    for(i = 0; i < n; i++)
    {
        if(infix[i] >= 'A' && infix[i] <= 'Z' ||
           infix[i] >= 'a' && infix[i] <= 'z' ||
           infix[i] >= '0' && infix[i] <= '9')
            prefix[j++] = infix[i];

        else if(infix[i] == '(')
            push(infix[i]);

        else if(infix[i] == ')')
        {
            while(top != -1 && stack[top] != '(')
                prefix[j++] = pop();

            if(top != -1)
                pop();
        }
        else
        {
            while(top != -1 && stack[top] != '(' &&
                  priority(stack[top]) > priority(infix[i]))
                prefix[j++] = pop();

            push(infix[i]);
        }
    }

    while(top != -1)
        prefix[j++] = pop();

    prefix[j] = '\0';

    for(i = 0; i < j/2; i++)
    {
        temp = prefix[i];
        prefix[i] = prefix[j-i-1];
        prefix[j-i-1] = temp;
    }

    printf("Prefix expression: %s", prefix);

    return 0;
}
