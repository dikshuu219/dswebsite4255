Q5. Enter the string from the user and reverse the string using stacks.

Program

#include <stdio.h>
#include <string.h>

int main()
{
    char str[100], stack[100];
    int top = -1, i;

    printf("Enter a string: ");
    scanf("%s", str);

    for(i = 0; str[i] != '\0'; i++)
    {
        top++;
        stack[top] = str[i];
    }

    printf("Reverse string: ");

    while(top != -1)
    {
        printf("%c", stack[top]);
        top--;
    }

    return 0;
}

Output

Enter a string: hello
Reverse string: olleh
