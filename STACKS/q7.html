Q7. Create two integer stacks in a single array.

Program

#include <stdio.h>

int a[10];
int top1 = -1, top2 = 10;

void push1()
{
    int x;

    if(top1 + 1 == top2)
        printf("Array is Full");
    else
    {
        printf("Enter number: ");
        scanf("%d", &x);
        top1++;
        a[top1] = x;
    }
}

void push2()
{
    int x;

    if(top1 + 1 == top2)
        printf("Array is Full");
    else
    {
        printf("Enter number: ");
        scanf("%d", &x);
        top2--;
        a[top2] = x;
    }
}

void pop1()
{
    if(top1 == -1)
        printf("Stack 1 is Empty");
    else
    {
        printf("Deleted: %d", a[top1]);
        top1--;
    }
}

void pop2()
{
    if(top2 == 10)
        printf("Stack 2 is Empty");
    else
    {
        printf("Deleted: %d", a[top2]);
        top2++;
    }
}

int main()
{
    int ch;

    while(1)
    {
        printf("\n\n1. Push Stack 1");
        printf("\n2. Pop Stack 1");
        printf("\n3. Push Stack 2");
        printf("\n4. Pop Stack 2");
        printf("\n5. Exit");
        printf("\nEnter choice: ");
        scanf("%d", &ch);

        if(ch == 1)
            push1();
        else if(ch == 2)
            pop1();
        else if(ch == 3)
            push2();
        else if(ch == 4)
            pop2();
        else if(ch == 5)
            break;
        else
            printf("Wrong choice");
    }

    return 0;
}
