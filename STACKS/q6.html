Q6. Create the program to match the single bracket using stacks.

Program

#include <stdio.h>

int main()
{
    char str[100], stack[100];
    int top = -1, i;

    printf("Enter expression: ");
    scanf("%s", str);

    for(i = 0; str[i] != '\0'; i++)
    {
        if(str[i] == '(')
        {
            top++;
            stack[top] = str[i];
        }
        else if(str[i] == ')')
        {
            if(top == -1)
            {
                printf("Brackets are not matched");
                return 0;
            }
            else
                top--;
        }
    }

    if(top == -1)
        printf("Brackets are matched");
    else
        printf("Brackets are not matched");

    return 0;
}

Output

Enter expression: ((a+b)*c)
Brackets are matched
