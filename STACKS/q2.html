Q2. Create a stack implementing the operations on the characters array (push, pop, peek, isempty, isfull)

Program

#include <stdio.h>

char a[5];
int top = -1;

void push()
{
    char x;

    if(top == 4)
        printf("Stack is Full");
    else
    {
        printf("Enter character: ");
        scanf(" %c", &x);
        top++;
        a[top] = x;
    }
}

void pop()
{
    if(top == -1)
        printf("Stack is Empty");
    else
    {
        printf("Deleted: %c", a[top]);
        top--;
    }
}

void peek()
{
    if(top == -1)
        printf("Stack is Empty");
    else
        printf("Top character: %c", a[top]);
}

void isempty()
{
    if(top == -1)
        printf("Stack is Empty");
    else
        printf("Stack is not Empty");
}

void isfull()
{
    if(top == 4)
        printf("Stack is Full");
    else
        printf("Stack is not Full");
}

int main()
{
    int ch;

    while(1)
    {
        printf("\n\n1. Push");
        printf("\n2. Pop");
        printf("\n3. Peek");
        printf("\n4. IsEmpty");
        printf("\n5. IsFull");
        printf("\n6. Exit");
        printf("\nEnter choice: ");
        scanf("%d", &ch);

        if(ch == 1)
            push();
        else if(ch == 2)
            pop();
        else if(ch == 3)
            peek();
        else if(ch == 4)
            isempty();
        else if(ch == 5)
            isfull();
        else if(ch == 6)
            break;
        else
            printf("Wrong choice");
    }

    return 0;
}
