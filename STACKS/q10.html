Q10. Enter the infix expression from user and convert into postfix expression.

Program

#include <stdio.h>

char stack[100];
int top = -1;

void push(char x)
{
    stack[++top] = x;
}

char pop()
{
    return stack[top--];
}

int priority(char x)
{
    if(x == '*' || x == '/')
        return 2;
    if(x == '+' || x == '-')
        return 1;
    return 0;
}

int main()
{
    char infix[100], postfix[100];
    int i, j = 0;

    printf("Enter infix expression: ");
    scanf("%s", infix);

    for(i = 0; infix[i] != '\0'; i++)
    {
        if((infix[i] >= 'A' && infix[i] <= 'Z') ||
           (infix[i] >= 'a' && infix[i] <= 'z') ||
           (infix[i] >= '0' && infix[i] <= '9'))
            postfix[j++] = infix[i];

        else if(infix[i] == '(')
            push(infix[i]);

        else if(infix[i] == ')')
        {
            while(top != -1 && stack[top] != '(')
                postfix[j++] = pop();

            if(top != -1)
                pop();
        }
        else
        {
            while(top != -1 && stack[top] != '(' &&
                  priority(stack[top]) >= priority(infix[i]))
                postfix[j++] = pop();

            push(infix[i]);
        }
    }

    while(top != -1)
        postfix[j++] = pop();

    postfix[j] = '\0';

    printf("Postfix expression: %s", postfix);

    return 0;
}

Postfix expression: AB+C*D-
