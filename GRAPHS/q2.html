Q2. Create the graph as a adjacency list with operations such as insertion, deletion

Program

#include <stdio.h>
#include <stdlib.h>

struct node
{
    int vertex;
    struct node *next;
};

struct node *a[10];

void insert(int x, int y)
{
    struct node *p;

    p = (struct node *)malloc(sizeof(struct node));
    p->vertex = y;
    p->next = a[x];
    a[x] = p;

    p = (struct node *)malloc(sizeof(struct node));
    p->vertex = x;
    p->next = a[y];
    a[y] = p;
}

void deleteEdge(int x, int y)
{
    struct node *p, *q;

    p = a[x];
    q = NULL;

    while (p != NULL)
    {
        if (p->vertex == y)
        {
            if (q == NULL)
                a[x] = p->next;
            else
                q->next = p->next;

            free(p);
            break;
        }

        q = p;
        p = p->next;
    }

    p = a[y];
    q = NULL;

    while (p != NULL)
    {
        if (p->vertex == x)
        {
            if (q == NULL)
                a[y] = p->next;
            else
                q->next = p->next;

            free(p);
            break;
        }

        q = p;
        p = p->next;
    }
}

void display(int n)
{
    int i;
    struct node *p;

    for (i = 0; i < n; i++)
    {
        printf("%d: ", i);
        p = a[i];

        while (p != NULL)
        {
            printf("%d -> ", p->vertex);
            p = p->next;
        }

        printf("NULL\n");
    }
}

int main()
{
    int n, ch, x, y, i;

    printf("Enter number of vertices: ");
    scanf("%d", &n);

    for (i = 0; i < n; i++)
        a[i] = NULL;

    do
    {
        printf("\n1. Insert Edge");
        printf("\n2. Delete Edge");
        printf("\n3. Display Graph");
        printf("\n4. Exit");
        printf("\nEnter choice: ");
        scanf("%d", &ch);

        if (ch == 1 || ch == 2)
        {
            printf("Enter two vertices: ");
            scanf("%d %d", &x, &y);

            if (x < 0 || x >= n || y < 0 || y >= n || x == y)
                printf("Invalid vertices\n");
            else if (ch == 1)
                insert(x, y);
            else
                deleteEdge(x, y);
        }
        else if (ch == 3)
            display(n);
        else if (ch != 4)
            printf("Wrong choice\n");

    } while (ch != 4);

    return 0;
}
