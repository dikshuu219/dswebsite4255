Q5. Traverse the adjacent matrix with breadth first search

Program

#include <stdio.h>

int main()
{
    int a[10][10] = {0};
    int visited[10] = {0};
    int queue[10];
    int n, e, i, x, y, start;
    int front = 0, rear = 0, v;

    printf("Enter number of vertices: ");
    scanf("%d", &n);

    printf("Enter number of edges: ");
    scanf("%d", &e);

    for (i = 0; i < e; i++)
    {
        printf("Enter edge: ");
        scanf("%d %d", &x, &y);

        a[x][y] = 1;
        a[y][x] = 1;
    }

    printf("Enter starting vertex: ");
    scanf("%d", &start);

    queue[rear++] = start;
    visited[start] = 1;

    printf("BFS Traversal: ");

    while (front < rear)
    {
        v = queue[front++];
        printf("%d ", v);

        for (i = 0; i < n; i++)
        {
            if (a[v][i] == 1 && visited[i] == 0)
            {
                queue[rear++] = i;
                visited[i] = 1;
            }
        }
    }

    return 0;
}

Output

Enter number of edges: 4
Enter edge: 0 1
Enter edge: 0 2
Enter edge: 1 2
Enter edge: 2 3
Enter starting vertex: 0
BFS Traversal: 0 1 2 3
