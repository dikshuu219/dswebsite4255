Q6. Traverse the adjacent list with breadth first search

Program

#include <stdio.h>
#include <stdlib.h>

struct node
{
    int vertex;
    struct node *next;
};

struct node *a[10];
int visited[10], queue[10];

void insert(int x, int y)
{
    struct node *p;

    p = (struct node *)malloc(sizeof(struct node));
    p->vertex = y;
    p->next = a[x];
    a[x] = p;

    p = (struct node *)malloc(sizeof(struct node));
    p->vertex = x;
    p->next = a[y];
    a[y] = p;
}

void BFS(int start, int n)
{
    int front = 0, rear = 0, v;
    struct node *p;

    queue[rear++] = start;
    visited[start] = 1;

    while (front < rear)
    {
        v = queue[front++];
        printf("%d ", v);

        p = a[v];

        while (p != NULL)
        {
            if (visited[p->vertex] == 0)
            {
                queue[rear++] = p->vertex;
                visited[p->vertex] = 1;
            }

            p = p->next;
        }
    }
}

int main()
{
    int n, e, i, x, y, start;

    printf("Enter number of vertices: ");
    scanf("%d", &n);

    for (i = 0; i < n; i++)
        a[i] = NULL;

    printf("Enter number of edges: ");
    scanf("%d", &e);

    for (i = 0; i < e; i++)
    {
        printf("Enter edge: ");
        scanf("%d %d", &x, &y);
        insert(x, y);
    }

    printf("Enter starting vertex: ");
    scanf("%d", &start);

    printf("BFS Traversal: ");
    BFS(start, n);

    return 0;
}
