Q1. Enter the vertices and edges and represent in adjacency matrix and print it.

Program

#include <stdio.h>

int main()
{
    int a[10][10], v, e, i, j, x, y;

    printf("Enter number of vertices: ");
    scanf("%d", &v);

    printf("Enter number of edges: ");
    scanf("%d", &e);

    for (i = 0; i < v; i++)
        for (j = 0; j < v; j++)
            a[i][j] = 0;

    for (i = 0; i < e; i++)
    {
        printf("Enter edge (two vertices): ");
        scanf("%d %d", &x, &y);

        a[x][y] = 1;
        a[y][x] = 1;
    }

    printf("Adjacency Matrix:\n");

    for (i = 0; i < v; i++)
    {
        for (j = 0; j < v; j++)
            printf("%d ", a[i][j]);

        printf("\n");
    }

    return 0;
}

Enter number of edges: 4

1 0 1 0
