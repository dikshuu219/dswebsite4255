Q4. Traverse the adjacent list with depth first search

Program

#include <stdio.h>
#include <stdlib.h>

struct node
{
    int vertex;
    struct node *next;
};

struct node *a[10];
int visited[10];

void insert(int x, int y)
{
    struct node *p;

    p = (struct node *)malloc(sizeof(struct node));
    p->vertex = y;
    p->next = a[x];
    a[x] = p;

    p = (struct node *)malloc(sizeof(struct node));
    p->vertex = x;
    p->next = a[y];
    a[y] = p;
}

void DFS(int v)
{
    struct node *p;

    printf("%d ", v);
    visited[v] = 1;

    p = a[v];

    while (p != NULL)
    {
        if (visited[p->vertex] == 0)
            DFS(p->vertex);

        p = p->next;
    }
}

int main()
{
    int n, e, i, x, y, start;

    printf("Enter number of vertices: ");
    scanf("%d", &n);

    for (i = 0; i < n; i++)
        a[i] = NULL;

    printf("Enter number of edges: ");
    scanf("%d", &e);

    for (i = 0; i < e; i++)
    {
        printf("Enter edge: ");
        scanf("%d %d", &x, &y);
        insert(x, y);
    }

    printf("Enter starting vertex: ");
    scanf("%d", &start);

    printf("DFS Traversal: ");
    DFS(start);

    return 0;
}
