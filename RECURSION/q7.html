Q7. Enter the number from the user and add all digits of the number using recursion.

Program

#include <stdio.h>

int sum(int n)
{
    if(n == 0)
        return 0;
    else
        return n % 10 + sum(n / 10);
}

int main()
{
    int n;

    printf("Enter a number: ");
    scanf("%d", &n);

    printf("Sum of digits = %d", sum(n));

    return 0;
}

Output

Enter a number: 1234
Sum of digits = 10
