Q8. Find the arithmetic mean of an integer array using recursion.

Program

#include <stdio.h>

int sum(int arr[], int n)
{
    if(n == 0)
        return 0;
    else
        return arr[n - 1] + sum(arr, n - 1);
}

int main()
{
    int arr[100], n, i;
    float mean;

    printf("Enter size of array: ");
    scanf("%d", &n);

    printf("Enter array elements: ");
    for(i = 0; i < n; i++)
        scanf("%d", &arr[i]);

    mean = (float)sum(arr, n) / n;

    printf("Arithmetic Mean = %.2f", mean);

    return 0;
}
