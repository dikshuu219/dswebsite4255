Q4. Convert a number into binary using recursion

Program

#include <stdio.h>

void binary(int n)
{
    if(n > 1)
        binary(n / 2);

    printf("%d", n % 2);
}

int main()
{
    int n;

    printf("Enter a number: ");
    scanf("%d", &n);

    printf("Binary number = ");
    binary(n);

    return 0;
}

Output

Enter a number: 10
Binary number = 1010
