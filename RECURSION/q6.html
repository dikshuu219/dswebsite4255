Q6. Find the smallest element in the integer array using recursion.

Program

#include <stdio.h>

int smallest(int arr[], int n)
{
    if(n == 1)
        return arr[0];

    int min = smallest(arr, n - 1);

    if(arr[n - 1] < min)
        return arr[n - 1];
    else
        return min;
}

int main()
{
    int arr[100], n, i;

    printf("Enter size of array: ");
    scanf("%d", &n);

    printf("Enter array elements: ");
    for(i = 0; i < n; i++)
        scanf("%d", &arr[i]);

    printf("Smallest element = %d", smallest(arr, n));

    return 0;
}
