Q5. Find the largest element in the integer array using recursion.

Program

#include <stdio.h>

int largest(int arr[], int n)
{
    if(n == 1)
        return arr[0];

    int max = largest(arr, n - 1);

    if(arr[n - 1] > max)
        return arr[n - 1];
    else
        return max;
}

int main()
{
    int arr[100], n, i;

    printf("Enter size of array: ");
    scanf("%d", &n);

    printf("Enter array elements: ");
    for(i = 0; i < n; i++)
        scanf("%d", &arr[i]);

    printf("Largest element = %d", largest(arr, n));

    return 0;
}

19
5
Largest element = 19
