Q6. Enter the Matrix from user, multiply it with a scalar value and display the answer?

Program

#include <stdio.h>

int main()
{
    int a[4][2], i, j, s;

    printf("Enter matrix elements:\n");

    for(i = 0; i < 4; i++)
    {
        for(j = 0; j < 2; j++)
        {
            scanf("%d", &a[i][j]);
        }
    }

    printf("Enter scalar value: ");
    scanf("%d", &s);

    for(i = 0; i < 4; i++)
    {
        for(j = 0; j < 2; j++)
        {
            a[i][j] = a[i][j] * s;
        }
    }

    printf("Answer matrix is:\n");

    for(i = 0; i < 4; i++)
    {
        for(j = 0; j < 2; j++)
        {
            printf("%d ", a[i][j]);
        }
        printf("\n");
    }

    return 0;
}
