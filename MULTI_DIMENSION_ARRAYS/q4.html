Q4. Enter two Matrices from user, multiply the matrices and display the answer?

Program

#include <stdio.h>

int main()
{
    int a[4][2], b[2][4], c[4][4];
    int i, j, k;

    printf("Enter first matrix (4x2):\n");
    for(i = 0; i < 4; i++)
    {
        for(j = 0; j < 2; j++)
            scanf("%d", &a[i][j]);
    }

    printf("Enter second matrix (2x4):\n");
    for(i = 0; i < 2; i++)
    {
        for(j = 0; j < 4; j++)
            scanf("%d", &b[i][j]);
    }

    for(i = 0; i < 4; i++)
    {
        for(j = 0; j < 4; j++)
        {
            c[i][j] = 0;

            for(k = 0; k < 2; k++)
                c[i][j] = c[i][j] + a[i][k] * b[k][j];
        }
    }

    printf("Multiplication of matrices:\n");
    for(i = 0; i < 4; i++)
    {
        for(j = 0; j < 4; j++)
            printf("%d ", c[i][j]);

        printf("\n");
    }

    return 0;
}
