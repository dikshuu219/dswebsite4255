Q5. Enter a Matrix from user, copy it in another matrix of same size?

Program

#include <stdio.h>

int main()
{
    int a[4][2], b[4][2];
    int i, j;

    printf("Enter matrix elements:\n");

    for(i = 0; i < 4; i++)
    {
        for(j = 0; j < 2; j++)
        {
            scanf("%d", &a[i][j]);
        }
    }

    for(i = 0; i < 4; i++)
    {
        for(j = 0; j < 2; j++)
        {
            b[i][j] = a[i][j];
        }
    }

    printf("Copied matrix is:\n");

    for(i = 0; i < 4; i++)
    {
        for(j = 0; j < 2; j++)
        {
            printf("%d ", b[i][j]);
        }
        printf("\n");
    }

    return 0;
}
