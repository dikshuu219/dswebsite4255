Q1. Enter a matrix of 4 rows and 2 columns from the user display it in matrix format?

Program

#include <stdio.h>

int main()
{
    int a[4][2], i, j;

    printf("Enter matrix elements:\n");

    for(i = 0; i < 4; i++)
    {
        for(j = 0; j < 2; j++)
        {
            scanf("%d", &a[i][j]);
        }
    }

    printf("Matrix is:\n");

    for(i = 0; i < 4; i++)
    {
        for(j = 0; j < 2; j++)
        {
            printf("%d ", a[i][j]);
        }
        printf("\n");
    }

    return 0;
}

Output

Enter matrix elements:
1
2
3
4
5
6
7
8
Matrix is:
1 2 
3 4 
5 6 
7 8
