Q7. Create a tic tac toe game using 2D Arrays.

Program

#include <stdio.h>

int main()
{
    char a[3][3] = {
        {'1', '2', '3'},
        {'4', '5', '6'},
        {'7', '8', '9'}
    };

    int i, j, choice, row, col, moves = 0;
    char player = 'X';
    int win = 0;

    while(moves < 9)
    {
        printf("\n");

        for(i = 0; i < 3; i++)
        {
            for(j = 0; j < 3; j++)
            {
                printf(" %c ", a[i][j]);

                if(j < 2)
                    printf("|");
            }
            printf("\n");

            if(i < 2)
                printf("---|---|---\n");
        }

        printf("\nPlayer %c, enter position (1-9): ", player);
        scanf("%d", &choice);

        if(choice < 1 || choice > 9)
        {
            printf("Invalid position! Try again.\n");
            continue;
        }

        row = (choice - 1) / 3;
        col = (choice - 1) % 3;

        if(a[row][col] == 'X' || a[row][col] == 'O')
        {
            printf("Position already taken! Try again.\n");
            continue;
        }

        a[row][col] = player;
        moves++;

        for(i = 0; i < 3; i++)
        {
            if(a[i][0] == player &&
               a[i][1] == player &&
               a[i][2] == player)
                win = 1;

            if(a[0][i] == player &&
               a[1][i] == player &&
               a[2][i] == player)
                win = 1;
        }

        if(a[0][0] == player &&
           a[1][1] == player &&
           a[2][2] == player)
            win = 1;

        if(a[0][2] == player &&
           a[1][1] == player &&
           a[2][0] == player)
            win = 1;

        if(win == 1)
        {
            printf("\nPlayer %c wins!\n", player);
            break;
        }

        if(player == 'X')
            player = 'O';
        else
            player = 'X';
    }

    if(win == 0)
        printf("\nMatch Draw!\n");

    printf("\nFinal Board:\n");

    for(i = 0; i < 3; i++)
    {
        for(j = 0; j < 3; j++)
        {
            printf(" %c ", a[i][j]);

            if(j < 2)
                printf("|");
        }
        printf("\n");

        if(i < 2)
            printf("---|---|---\n");
    }

    return 0;
}
