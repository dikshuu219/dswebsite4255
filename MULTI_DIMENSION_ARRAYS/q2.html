Q2. Enter two Matrices from user, add the matrices and display the answer?

Program

#include <stdio.h>

int main()
{
    int a[4][2], b[4][2], c[4][2];
    int i, j;

    printf("Enter first matrix:\n");
    for(i = 0; i < 4; i++)
    {
        for(j = 0; j < 2; j++)
            scanf("%d", &a[i][j]);
    }

    printf("Enter second matrix:\n");
    for(i = 0; i < 4; i++)
    {
        for(j = 0; j < 2; j++)
            scanf("%d", &b[i][j]);
    }

    for(i = 0; i < 4; i++)
    {
        for(j = 0; j < 2; j++)
            c[i][j] = a[i][j] + b[i][j];
    }

    printf("Sum of matrices:\n");
    for(i = 0; i < 4; i++)
    {
        for(j = 0; j < 2; j++)
            printf("%d ", c[i][j]);

        printf("\n");
    }

    return 0;
}
