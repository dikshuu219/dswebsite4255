Q20. Enter the string from the user and count the number of characters, words, spaces and lines

Program

#include <stdio.h>

int main()
{
    char str[500];
    int i, characters = 0, words = 0;
    int spaces = 0, lines = 1, inWord = 0;

    printf("Enter a string:\n");
    fgets(str, sizeof(str), stdin);

    for(i = 0; str[i] != '\0'; i++)
    {
        if(str[i] != '\n')
            characters++;

        if(str[i] == ' ')
            spaces++;

        if(str[i] == '\n')
            lines++;

        if(str[i] != ' ' && str[i] != '\n' && inWord == 0)
        {
            words++;
            inWord = 1;
        }
        else if(str[i] == ' ' || str[i] == '\n')
        {
            inWord = 0;
        }
    }

    printf("Characters = %d\n", characters);
    printf("Words = %d\n", words);
    printf("Spaces = %d\n", spaces);
    printf("Lines = %d\n", lines);

    return 0;
}

Output

Enter a string:
Hello World
Characters = 11
Words = 2
Spaces = 1
Lines = 2
