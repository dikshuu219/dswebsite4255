Q19. Enter the string from the user and convert into a float value?

Program

#include <stdio.h>
#include <stdlib.h>

int main()
{
    char str[100];
    float num;

    printf("Enter a string: ");
    scanf("%99s", str);

    num = atof(str);

    printf("Float value = %.2f", num);

    return 0;
}

Output

Enter a string: 3.14
Float value = 3.14
