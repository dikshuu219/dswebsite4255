Q28. Create a dictionary having 10 words and their meaning with following operations. [create, delete, search].

Program

#include <stdio.h>
#include <string.h>

int main()
{
    char word[10][20] = {
        "Apple", "Book", "Cat", "Dog", "Easy",
        "Friend", "Happy", "Kind", "Love", "School"
    };

    char meaning[10][50] = {
        "A fruit", "Something to read", "An animal",
        "A pet animal", "Not difficult", "A person we like",
        "Feeling of joy", "Being helpful", "A strong feeling",
        "A place to study"
    };

    char search[20];
    int i, choice, found;

    printf("1. Create Dictionary\n");
    printf("2. Search Word\n");
    printf("3. Delete Word\n");
    printf("Enter your choice: ");
    scanf("%d", &choice);

    if(choice == 1)
    {
        printf("Dictionary created successfully.\n");

        for(i = 0; i < 10; i++)
            printf("%s - %s\n", word[i], meaning[i]);
    }
    else if(choice == 2 || choice == 3)
    {
        printf("Enter word: ");
        scanf("%19s", search);

        found = 0;

        for(i = 0; i < 10; i++)
        {
            if(strcmp(word[i], search) == 0)
            {
                found = 1;

                if(choice == 2)
                    printf("Meaning: %s\n", meaning[i]);
                else
                {
                    strcpy(word[i], "");
                    strcpy(meaning[i], "");
                    printf("Word deleted successfully.\n");
                }

                break;
            }
        }

        if(found == 0)
            printf("Word not found.\n");
    }
    else
        printf("Invalid choice.");

    return 0;
}
