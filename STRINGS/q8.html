Q8. Enter two strings from user and concatenate those strings in another string and display the answer?

Program

#include <stdio.h>

int main()
{
    char str1[100], str2[100], str3[200];
    int i, j;

    printf("Enter first string: ");
    fgets(str1, sizeof(str1), stdin);

    printf("Enter second string: ");
    fgets(str2, sizeof(str2), stdin);

    for(i = 0; str1[i] != '\0' && str1[i] != '\n'; i++)
    {
        str3[i] = str1[i];
    }

    for(j = 0; str2[j] != '\0' && str2[j] != '\n'; j++)
    {
        str3[i + j] = str2[j];
    }

    str3[i + j] = '\0';

    printf("Concatenated string = %s", str3);

    return 0;
}

Output

Enter first string: Data 
Enter second string: Structure
Concatenated string = Data Structure
