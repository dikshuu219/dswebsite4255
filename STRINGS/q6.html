Q6. Enter the string from user and final length of the string ?

Program

#include <stdio.h>

int main()
{
    char str[100];
    int i, length = 0;

    printf("Enter a string: ");
    fgets(str, sizeof(str), stdin);

    for(i = 0; str[i] != '\0' && str[i] != '\n'; i++)
    {
        length++;
    }

    printf("Length of string = %d", length);

    return 0;
}

Output

Enter a string: Data Structure
Length of string = 14
