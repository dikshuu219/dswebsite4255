Q14. Enter the string from the user and print the difference of ASCII value of each character ?

Program

#include <stdio.h>

int main()
{
    char str[100];
    int i;

    printf("Enter a string: ");
    fgets(str, sizeof(str), stdin);

    for(i = 0; str[i] != '\0' && str[i] != '\n'; i++)
    {
        if(str[i + 1] != '\0' && str[i + 1] != '\n')
        {
            printf("Difference between %c and %c = %d\n",
                   str[i], str[i + 1],
                   str[i + 1] - str[i]);
        }
    }

    return 0;
}

Output

Enter a string: abcd
Difference between a and b = 1
Difference between b and c = 1
Difference between c and d = 1
