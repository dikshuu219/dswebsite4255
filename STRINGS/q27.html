Q27. Display the string on the output screen but ask the text alignment option given by the user (1 for left, 2 for center, 3 right)

Program

#include <stdio.h>
#include <string.h>

int main()
{
    char str[100];
    int choice, i, space;

    printf("Enter a string: ");
    scanf(" %[^\n]", str);

    printf("1. Left Alignment\n");
    printf("2. Center Alignment\n");
    printf("3. Right Alignment\n");
    printf("Enter your choice: ");
    scanf("%d", &choice);

    space = 30 - strlen(str);

    if(choice == 1)
    {
        printf("%s", str);
    }
    else if(choice == 2)
    {
        for(i = 0; i < space / 2; i++)
            printf(" ");

        printf("%s", str);
    }
    else if(choice == 3)
    {
        for(i = 0; i < space; i++)
            printf(" ");

        printf("%s", str);
    }
    else
    {
        printf("Invalid choice");
    }

    return 0;
}

Output

Enter a string: hello world
1. Left Alignment
2. Center Alignment
3. Right Alignment
Enter your choice: 2
         hello world
