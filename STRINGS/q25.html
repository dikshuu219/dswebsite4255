Q25. Enter the string from the user and trim/ eliminate the spaces from both ends (if any).

Program

#include <stdio.h>
#include <string.h>

int main()
{
    char str[100];
    int i, start = 0, end;

    printf("Enter a string: ");
    scanf(" %[^\n]", str);

    end = strlen(str) - 1;

    while(str[start] == ' ')
        start++;

    while(end >= start && str[end] == ' ')
        end--;

    printf("String after trimming: ");

    for(i = start; i <= end; i++)
        printf("%c", str[i]);

    return 0;
}

Output

Enter a string:    hello world   
String after trimming: hello world
