Q13. Enter the string from the user and copy reverse of the string into another string ?

Program

#include <stdio.h>

int main()
{
    char str1[100], str2[100];
    int i, length = 0;

    printf("Enter a string: ");
    fgets(str1, sizeof(str1), stdin);

    while(str1[length] != '\0' && str1[length] != '\n')
    {
        length++;
    }

    for(i = 0; i < length; i++)
    {
        str2[i] = str1[length - i - 1];
    }

    str2[length] = '\0';

    printf("Reversed string = %s", str2);

    return 0;
}

Output

Enter a string: hello
Reversed string = olleh
