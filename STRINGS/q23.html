Q23. Enter a sentence from user and then enter the word and count the occurence of that word in the sentence.

Program

#include <stdio.h>
#include <string.h>

int main()
{
    char sentence[200], word[50];
    int i, j, count = 0;

    printf("Enter a sentence: ");
    fgets(sentence, sizeof(sentence), stdin);

    printf("Enter a word: ");
    scanf("%s", word);

    for(i = 0; sentence[i] != '\0'; i++)
    {
        for(j = 0; word[j] != '\0'; j++)
        {
            if(sentence[i + j] != word[j])
                break;
        }

        if(word[j] == '\0')
            count++;
    }

    printf("Occurrence of word = %d", count);

    return 0;
}

Output

Enter a sentence: the cat and the dog and the bird
Enter a word: the
Occurrence of word = 3
