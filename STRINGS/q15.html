Q15. Enter the string from the user, enter a character also and search the character in the string?

Program

#include <stdio.h>

int main()
{
    char str[100], ch;
    int i, found = 0;

    printf("Enter a string: ");
    fgets(str, sizeof(str), stdin);

    printf("Enter a character to search: ");
    scanf("%c", &ch);

    for(i = 0; str[i] != '\0' && str[i] != '\n'; i++)
    {
        if(str[i] == ch)
        {
            found = 1;
            break;
        }
    }

    if(found == 1)
    {
        printf("Character found in the string");
    }
    else
    {
        printf("Character not found in the string");
    }

    return 0;
}

Character found in the string
