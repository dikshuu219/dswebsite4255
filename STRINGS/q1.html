Q1. Enter the string from user and convert in uppercase ?

Program

#include <stdio.h>

int main()
{
    char str[100];
    int i;

    printf("Enter a string: ");
    gets(str);

    for(i = 0; str[i] != '\0'; i++)
    {
        if(str[i] >= 'a' && str[i] <= 'z')
        {
            str[i] = str[i] - 32;
        }
    }

    printf("Uppercase string = %s", str);

    return 0;
}

Output

Enter a string: hello world
Uppercase string = HELLO WORLD
