Q18. Enter the string from the user and convert into an integer value?

Program

#include <stdio.h>

int main()
{
    char str[100];
    int i, num = 0, sign = 1;

    printf("Enter a numeric string: ");
    scanf("%99s", str);

    i = 0;

    if(str[i] == '-')
    {
        sign = -1;
        i++;
    }

    for(; str[i] != '\0'; i++)
    {
        if(str[i] >= '0' && str[i] <= '9')
        {
            num = num * 10 + (str[i] - '0');
        }
        else
        {
            printf("Invalid input");
            return 0;
        }
    }

    num = num * sign;

    printf("Integer value = %d", num);

    return 0;
}

Output

Enter a numeric string: 12345
Integer value = 12345
