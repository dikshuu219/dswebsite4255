Q24. Enter a sentence from user and then enter the word and replace that word by another word.

Program

#include <stdio.h>
#include <string.h>

int main()
{
    char sentence[200], word[50], replace[50];
    char result[300];
    int i, j, k = 0, found;

    printf("Enter a sentence: ");
    fgets(sentence, sizeof(sentence), stdin);

    printf("Enter the word to replace: ");
    scanf("%49s", word);

    printf("Enter the new word: ");
    scanf("%49s", replace);

    for(i = 0; sentence[i] != '\0'; i++)
    {
        found = 1;

        for(j = 0; word[j] != '\0'; j++)
        {
            if(sentence[i + j] != word[j])
            {
                found = 0;
                break;
            }
        }

        if(found == 1 && word[j] == '\0')
        {
            for(j = 0; replace[j] != '\0'; j++)
                result[k++] = replace[j];

            i = i + strlen(word) - 1;
        }
        else
        {
            result[k++] = sentence[i];
        }
    }

    result[k] = '\0';

    printf("New sentence: %s", result);

    return 0;
}

Output

Enter a sentence: I like apple and apple
Enter the word to replace: apple
Enter the new word: mango
New sentence: I like mango and mango
