Q10. Enter the string from user, enter the index number and print character present on that index ?

Program

#include <stdio.h>

int main()
{
    char str[100];
    int index, i, length = 0;

    printf("Enter a string: ");
    fgets(str, sizeof(str), stdin);

    while(str[length] != '\0' && str[length] != '\n')
    {
        length++;
    }

    printf("Enter index number: ");
    scanf("%d", &index);

    if(index >= 0 && index < length)
    {
        printf("Character at index %d = %c", index, str[index]);
    }
    else
    {
        printf("Invalid index number");
    }

    return 0;
}

Output

Enter a string: computer
Enter index number: 3
Character at index 3 = p
