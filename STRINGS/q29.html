Q29. Enter the string from user and convert characters into their ASCII values in the integer array. [for example take 10 characters]

Program

#include <stdio.h>

int main()
{
    char str[11];
    int arr[10], i;

    printf("Enter 10 characters: ");
    scanf("%10s", str);

    for(i = 0; i < 10; i++)
    {
        arr[i] = str[i];
    }

    printf("ASCII values are:\n");

    for(i = 0; i < 10; i++)
    {
        printf("%d ", arr[i]);
    }

    return 0;
}

Output

Enter 10 characters: abcdefghij
ASCII values are:
97 98 99 100 101 102 103 104 105 106
