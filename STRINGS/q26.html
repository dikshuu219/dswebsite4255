Q26. Enter the string from user and splits the first word and second word and return the start index of the second word.

Program

#include <stdio.h>

int main()
{
    char str[100];
    int i = 0, start;

    printf("Enter a string: ");
    scanf(" %[^\n]", str);

    while(str[i] == ' ')
        i++;

    while(str[i] != ' ' && str[i] != '\0')
        i++;

    while(str[i] == ' ')
        i++;

    start = i;

    printf("First word: ");
    for(i = 0; str[i] != ' ' && str[i] != '\0'; i++)
        printf("%c", str[i]);

    printf("\nSecond word: ");
    for(i = start; str[i] != ' ' && str[i] != '\0'; i++)
        printf("%c", str[i]);

    printf("\nStarting index of second word: %d", start);

    return 0;
}

Output

Enter a string:   hello   world
First word: hello
Second word: world
Starting index of second word: 8
