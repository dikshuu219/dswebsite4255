Q12. Enter an array from user and remove the duplicate values and fill with zero?

Program

#include <stdio.h>

int main()
{
    int a[100], n, i, j;

    printf("Enter size of array: ");
    scanf("%d", &n);

    printf("Enter array elements: ");
    for(i = 0; i < n; i++)
    {
        scanf("%d", &a[i]);
    }

    for(i = 0; i < n; i++)
    {
        for(j = i + 1; j < n; j++)
        {
            if(a[i] == a[j])
            {
                a[j] = 0;
            }
        }
    }

    printf("Array after removing duplicates: ");
    for(i = 0; i < n; i++)
    {
        printf("%d ", a[i]);
    }

    return 0;
}
