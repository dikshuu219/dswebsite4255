Q17. Merge the two sorted arrays into one array but in sorted order?

Program

#include <stdio.h>

int main()
{
    int a[100], b[100], c[200];
    int n, m, i, j, k;

    printf("Enter size of first array: ");
    scanf("%d", &n);

    printf("Enter elements of first sorted array: ");
    for(i = 0; i < n; i++)
    {
        scanf("%d", &a[i]);
    }

    printf("Enter size of second array: ");
    scanf("%d", &m);

    printf("Enter elements of second sorted array: ");
    for(i = 0; i < m; i++)
    {
        scanf("%d", &b[i]);
    }

    i = 0;
    j = 0;
    k = 0;

    while(i < n && j < m)
    {
        if(a[i] < b[j])
        {
            c[k] = a[i];
            i++;
        }
        else
        {
            c[k] = b[j];
            j++;
        }
        k++;
    }

    while(i < n)
    {
        c[k] = a[i];
        i++;
        k++;
    }

    while(j < m)
    {
        c[k] = b[j];
        j++;
        k++;
    }

    printf("Merged array in sorted order: ");
    for(i = 0; i < n + m; i++)
    {
        printf("%d ", c[i]);
    }

    return 0;
}

Output

9

8
Merged array in sorted order: 1 2 4 6 8 9
