Q3. Enter an array from user and copy the array into second array?

Program

#include <stdio.h>

int main()
{
    int a[100], b[100], n, i;

    printf("Enter size of array: ");
    scanf("%d", &n);

    printf("Enter array elements: ");
    for(i = 0; i < n; i++)
    {
        scanf("%d", &a[i]);
    }

    for(i = 0; i < n; i++)
    {
        b[i] = a[i];
    }

    printf("Elements of second array are: ");
    for(i = 0; i < n; i++)
    {
        printf("%d ", b[i]);
    }

    return 0;
}

Output

Enter size of array: 5
Enter array elements: 12
7
3
19
5
Elements of second array are: 12 7 3 19 5
