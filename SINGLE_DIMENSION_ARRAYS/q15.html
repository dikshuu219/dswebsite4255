Q15. Enter an array from user and print all the palindrome numbers?

Program

#include <stdio.h>

int main()
{
    int a[100], n, i, num, rem, rev;

    printf("Enter size of array: ");
    scanf("%d", &n);

    printf("Enter array elements: ");
    for(i = 0; i < n; i++)
    {
        scanf("%d", &a[i]);
    }

    printf("Palindrome numbers are: ");

    for(i = 0; i < n; i++)
    {
        num = a[i];
        rev = 0;

        while(num > 0)
        {
            rem = num % 10;
            rev = rev * 10 + rem;
            num = num / 10;
        }

        if(a[i] == rev && a[i] >= 0)
        {
            printf("%d ", a[i]);
        }
    }

    return 0;
}

Output

Enter array elements: 121
45
33
7
1001
56
Palindrome numbers are: 121 33 7 1001
