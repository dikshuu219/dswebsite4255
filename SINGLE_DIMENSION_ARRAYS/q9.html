Q9. Enter an array from user and find the k-th smallest value?

Program

#include <stdio.h>

int main()
{
    int a[100], n, i, j, temp, k;

    printf("Enter size of array: ");
    scanf("%d", &n);

    printf("Enter array elements: ");
    for(i = 0; i < n; i++)
    {
        scanf("%d", &a[i]);
    }

    printf("Enter the value of k: ");
    scanf("%d", &k);

    for(i = 0; i < n - 1; i++)
    {
        for(j = 0; j < n - i - 1; j++)
        {
            if(a[j] > a[j + 1])
            {
                temp = a[j];
                a[j] = a[j + 1];
                a[j + 1] = temp;
            }
        }
    }

    if(k >= 1 && k <= n)
    {
        printf("K-th smallest value = %d", a[k - 1]);
    }
    else
    {
        printf("Invalid value of k");
    }

    return 0;
}
