Q20. Trapping Rain Water Problem: Given n non-negative integers representing an elevation map where the width of each bar is 1, compute how much water it can trap after raining.

Program

#include <stdio.h>

int main()
{
    int a[100], n, i, j;
    int left, right, water = 0;

    printf("Enter number of bars: ");
    scanf("%d", &n);

    printf("Enter heights of bars: ");
    for(i = 0; i < n; i++)
    {
        scanf("%d", &a[i]);
    }

    for(i = 1; i < n - 1; i++)
    {
        left = a[i];
        right = a[i];

        for(j = 0; j < i; j++)
        {
            if(a[j] > left)
                left = a[j];
        }

        for(j = i + 1; j < n; j++)
        {
            if(a[j] > right)
                right = a[j];
        }

        if(left < right)
            water = water + left - a[i];
        else
            water = water + right - a[i];
    }

    printf("Trapped water = %d", water);

    return 0;
}
