Q7. Enter an array from user and square all the values of array ?

Program

#include <stdio.h>

int main()
{
    int a[100], n, i;

    printf("Enter size of array: ");
    scanf("%d", &n);

    printf("Enter array elements: ");
    for(i = 0; i < n; i++)
    {
        scanf("%d", &a[i]);
    }

    printf("Square of array elements: ");
    for(i = 0; i < n; i++)
    {
        printf("%d ", a[i] * a[i]);
    }

    return 0;
}

Output

Enter size of array: 5
Enter array elements: 12
7
3
19
5
Square of array elements: 144 49 9 361 25
