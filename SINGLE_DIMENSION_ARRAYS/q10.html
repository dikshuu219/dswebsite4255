Q10. Enter an array from user and add all values ?

Program

#include <stdio.h>

int main()
{
    int a[100], n, i, sum = 0;

    printf("Enter size of array: ");
    scanf("%d", &n);

    printf("Enter array elements: ");
    for(i = 0; i < n; i++)
    {
        scanf("%d", &a[i]);
    }

    for(i = 0; i < n; i++)
    {
        sum = sum + a[i];
    }

    printf("Sum of array elements = %d", sum);

    return 0;
}

Output

Enter size of array: 5
Enter array elements: 12
7
3
19
5
Sum of array elements = 46
