Q8. Enter an array from user and find maximum and minimum value?

Program

#include <stdio.h>

int main()
{
    int a[100], n, i, max, min;

    printf("Enter size of array: ");
    scanf("%d", &n);

    printf("Enter array elements: ");
    for(i = 0; i < n; i++)
    {
        scanf("%d", &a[i]);
    }

    max = a[0];
    min = a[0];

    for(i = 1; i < n; i++)
    {
        if(a[i] > max)
        {
            max = a[i];
        }

        if(a[i] < min)
        {
            min = a[i];
        }
    }

    printf("Maximum value = %d\n", max);
    printf("Minimum value = %d\n", min);

    return 0;
}

Output

Enter array elements: 12
7
3
19
5
Maximum value = 19
Minimum value = 3
