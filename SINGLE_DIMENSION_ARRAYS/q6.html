Q6. Enter an array from user and print the number which is positive ?

Program

#include <stdio.h>

int main()
{
    int a[100], n, i;

    printf("Enter size of array: ");
    scanf("%d", &n);

    printf("Enter array elements: ");
    for(i = 0; i < n; i++)
    {
        scanf("%d", &a[i]);
    }

    printf("Positive numbers are: ");
    for(i = 0; i < n; i++)
    {
        if(a[i] > 0)
        {
            printf("%d ", a[i]);
        }
    }

    return 0;
}

Output

Enter size of array: 5
Enter array elements: 4
-3
7
-8
2
Positive numbers are: 4 7 2
