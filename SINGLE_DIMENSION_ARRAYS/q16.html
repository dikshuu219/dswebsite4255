Q16. Enter two arrays from the user and check they are same or not?

Program

#include <stdio.h>

int main()
{
    int a[100], b[100], n, m, i, same = 1;

    printf("Enter size of first array: ");
    scanf("%d", &n);

    printf("Enter elements of first array: ");
    for(i = 0; i < n; i++)
    {
        scanf("%d", &a[i]);
    }

    printf("Enter size of second array: ");
    scanf("%d", &m);

    printf("Enter elements of second array: ");
    for(i = 0; i < m; i++)
    {
        scanf("%d", &b[i]);
    }

    if(n != m)
    {
        same = 0;
    }
    else
    {
        for(i = 0; i < n; i++)
        {
            if(a[i] != b[i])
            {
                same = 0;
                break;
            }
        }
    }

    if(same == 1)
    {
        printf("Both arrays are same");
    }
    else
    {
        printf("Both arrays are not same");
    }

    return 0;
}
