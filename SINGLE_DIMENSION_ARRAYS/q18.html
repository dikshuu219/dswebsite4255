Q18. Find the local minima in the array? (An element in an array is a local minima if it less than the element after it, and the element before it)

Program

#include <stdio.h>

int main()
{
    int a[100], n, i;

    printf("Enter size of array: ");
    scanf("%d", &n);

    printf("Enter array elements: ");
    for(i = 0; i < n; i++)
    {
        scanf("%d", &a[i]);
    }

    printf("Local minima are: ");

    for(i = 1; i < n - 1; i++)
    {
        if(a[i] < a[i - 1] && a[i] < a[i + 1])
        {
            printf("%d ", a[i]);
        }
    }

    return 0;
}

Output

Enter array elements: 9
3
8
2
6
1
5
Local minima are: 3 2 1
