Q11. Enter an array from user and find the duplicate values?

Program

#include <stdio.h>

int main()
{
    int a[100], n, i, j;

    printf("Enter size of array: ");
    scanf("%d", &n);

    printf("Enter array elements: ");
    for(i = 0; i < n; i++)
    {
        scanf("%d", &a[i]);
    }

    printf("Duplicate values are: ");

    for(i = 0; i < n; i++)
    {
        for(j = i + 1; j < n; j++)
        {
            if(a[i] == a[j])
            {
                printf("%d ", a[i]);
                break;
            }
        }
    }

    return 0;
}

Output

Enter size of array: 6
Enter array elements: 4
7
4
2
7
9
Duplicate values are: 4 7
