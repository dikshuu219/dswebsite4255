Q9. Calculate the best and worst complexity of the linear searching algorithm with exact time taken by the algorithm on integer array.

Program

#include <stdio.h>
#include <time.h>

int main()
{
    int a[100], n, i, x, found = 0;
    clock_t start, end;
    double time_taken;

    printf("Enter number of elements: ");
    scanf("%d", &n);

    printf("Enter array elements: ");
    for (i = 0; i < n; i++)
        scanf("%d", &a[i]);

    printf("Enter value to search: ");
    scanf("%d", &x);

    start = clock();

    for (i = 0; i < n; i++)
    {
        if (a[i] == x)
        {
            found = 1;
            break;
        }
    }

    end = clock();

    time_taken = (double)(end - start) / CLOCKS_PER_SEC;

    if (found == 1)
        printf("Element found at position %d\n", i + 1);
    else
        printf("Element not found\n");

    printf("Best Case Complexity: O(1)\n");
    printf("Worst Case Complexity: O(n)\n");
    printf("Time taken: %f seconds\n", time_taken);

    return 0;
}
