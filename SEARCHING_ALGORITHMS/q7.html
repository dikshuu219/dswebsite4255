Q7. Enter a integer array from the user, enter an integer value and perform binary search algorithm through recursion.

Program

#include <stdio.h>

int search(int a[], int low, int high, int x)
{
    if (low > high)
        return -1;

    int mid = (low + high) / 2;

    if (a[mid] == x)
        return mid;
    else if (a[mid] < x)
        return search(a, mid + 1, high, x);
    else
        return search(a, low, mid - 1, x);
}

int main()
{
    int a[100], n, i, x, pos;

    printf("Enter number of elements: ");
    scanf("%d", &n);

    printf("Enter array elements in sorted order: ");
    for (i = 0; i < n; i++)
        scanf("%d", &a[i]);

    printf("Enter value to search: ");
    scanf("%d", &x);

    pos = search(a, 0, n - 1, x);

    if (pos == -1)
        printf("Element not found");
    else
        printf("Element found at position %d", pos + 1);

    return 0;
}
