Q2. Enter a string from the user, enter a character and perform linear search algorithm through recursion.

Program

#include <stdio.h>

int search(char s[], char ch, int i)
{
    if (s[i] == '\0')
        return -1;

    if (s[i] == ch)
        return i;

    return search(s, ch, i + 1);
}

int main()
{
    char s[100], ch;
    int pos;

    printf("Enter a string: ");
    scanf("%s", s);

    printf("Enter character to search: ");
    scanf(" %c", &ch);

    pos = search(s, ch, 0);

    if (pos == -1)
        printf("Character not found");
    else
        printf("Character found at position %d", pos + 1);

    return 0;
}
