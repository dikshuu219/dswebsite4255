Q10. Calculate the best and worst complexity of the binary searching algorithm with exact time taken by the algorithm on integer array.

Program

#include <stdio.h>
#include <time.h>

int main()
{
    int a[100], n, i, x;
    int low, high, mid, found = 0;
    clock_t start, end;
    double time_taken;

    printf("Enter number of elements: ");
    scanf("%d", &n);

    printf("Enter sorted array elements: ");
    for (i = 0; i < n; i++)
        scanf("%d", &a[i]);

    printf("Enter value to search: ");
    scanf("%d", &x);

    low = 0;
    high = n - 1;

    start = clock();

    while (low <= high)
    {
        mid = (low + high) / 2;

        if (a[mid] == x)
        {
            found = 1;
            break;
        }
        else if (a[mid] < x)
            low = mid + 1;
        else
            high = mid - 1;
    }

    end = clock();

    time_taken = (double)(end - start) / CLOCKS_PER_SEC;

    if (found == 1)
        printf("Element found at position %d\n", mid + 1);
    else
        printf("Element not found\n");

    printf("Best Case Complexity: O(1)\n");
    printf("Worst Case Complexity: O(log n)\n");
    printf("Time taken: %f seconds\n", time_taken);

    return 0;
}
