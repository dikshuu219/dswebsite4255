Q8. Enter a string from the user, enter a character and perform binary search algorithm through iteration.

Program

#include <stdio.h>
#include <string.h>

int main()
{
    char s[100], ch, temp;
    int i, j, low, high, mid, found = 0;

    printf("Enter a string: ");
    scanf("%s", s);

    for (i = 0; s[i] != '\0'; i++)
    {
        for (j = i + 1; s[j] != '\0'; j++)
        {
            if (s[i] > s[j])
            {
                temp = s[i];
                s[i] = s[j];
                s[j] = temp;
            }
        }
    }

    printf("Sorted string: %s\n", s);

    printf("Enter character to search: ");
    scanf(" %c", &ch);

    low = 0;
    high = strlen(s) - 1;

    while (low <= high)
    {
        mid = (low + high) / 2;

        if (s[mid] == ch)
        {
            printf("Character found at position %d", mid + 1);
            found = 1;
            break;
        }
        else if (s[mid] < ch)
            low = mid + 1;
        else
            high = mid - 1;
    }

    if (found == 0)
        printf("Character not found");

    return 0;
}
