Q6. Enter a string from the user, enter a character and perform binary search algorithm through recursion.

Program

#include <stdio.h>
#include <string.h>

int search(char s[], char ch, int low, int high)
{
    if (low > high)
        return -1;

    int mid = (low + high) / 2;

    if (s[mid] == ch)
        return mid;
    else if (s[mid] < ch)
        return search(s, ch, mid + 1, high);
    else
        return search(s, ch, low, mid - 1);
}

int main()
{
    char s[100], ch;
    int i, j, pos;
    char temp;

    printf("Enter a string in sorted order: ");
    scanf("%s", s);

    for (i = 0; s[i] != '\0'; i++)
    {
        for (j = i + 1; s[j] != '\0'; j++)
        {
            if (s[i] > s[j])
            {
                temp = s[i];
                s[i] = s[j];
                s[j] = temp;
            }
        }
    }

    printf("Sorted string: %s\n", s);

    printf("Enter character to search: ");
    scanf(" %c", &ch);

    pos = search(s, ch, 0, strlen(s) - 1);

    if (pos == -1)
        printf("Character not found");
    else
        printf("Character found at position %d", pos + 1);

    return 0;
}
