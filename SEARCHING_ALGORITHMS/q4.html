Q4. Enter a string from the user, enter a character and perform linear search algorithm through iteration.

Program

#include <stdio.h>

int main()
{
    char s[100], ch;
    int i, found = 0;

    printf("Enter a string: ");
    scanf("%s", s);

    printf("Enter character to search: ");
    scanf(" %c", &ch);

    for (i = 0; s[i] != '\0'; i++)
    {
        if (s[i] == ch)
        {
            printf("Character found at position %d", i + 1);
            found = 1;
            break;
        }
    }

    if (found == 0)
        printf("Character not found");

    return 0;
}

Output

Enter a string: education
Enter character to search: c
Character found at position 4
