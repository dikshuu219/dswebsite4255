Q18. Reverse First K elements of Queue

Program

#include <stdio.h>

int main()
{
    int a[100], stack[100];
    int n, k, i, top = -1;

    printf("Enter number of elements: ");
    scanf("%d", &n);

    printf("Enter queue elements: ");
    for(i = 0; i < n; i++)
        scanf("%d", &a[i]);

    printf("Enter K: ");
    scanf("%d", &k);

    if(k < 1 || k > n)
    {
        printf("Invalid K");
        return 0;
    }

    for(i = 0; i < k; i++)
        stack[++top] = a[i];

    for(i = 0; i < k; i++)
        a[i] = stack[top--];

    printf("Queue after reversing: ");
    for(i = 0; i < n; i++)
        printf("%d ", a[i]);

    return 0;
}

Output

Enter number of elements: 5
Enter queue elements: 1
2
3
4
5
Enter K: 3
Queue after reversing: 3 2 1 4 5
