Q13. Find the first circular tour that visits all petrol pumps

Program

#include <stdio.h>

int main()
{
    int petrol[10], distance[10];
    int n, i, start = 0, balance = 0, total = 0;

    printf("Enter number of petrol pumps: ");
    scanf("%d", &n);

    for(i = 0; i < n; i++)
    {
        printf("Enter petrol and distance: ");
        scanf("%d %d", &petrol[i], &distance[i]);
    }

    for(i = 0; i < n; i++)
    {
        balance = balance + petrol[i] - distance[i];
        total = total + petrol[i] - distance[i];

        if(balance < 0)
        {
            start = i + 1;
            balance = 0;
        }
    }

    if(total < 0)
        printf("Circular tour is not possible");
    else
        printf("First petrol pump: %d", start + 1);

    return 0;
}
