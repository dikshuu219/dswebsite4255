Q1. Implement the operations of simple queue in an integer array (insert, delete, peek, isempty, isfull)

Program

#include <stdio.h>

int a[5], front = -1, rear = -1;

void insert()
{
    int x;

    if(rear == 4)
        printf("Queue is Full");
    else
    {
        printf("Enter number: ");
        scanf("%d", &x);

        if(front == -1)
            front = 0;

        rear++;
        a[rear] = x;
    }
}

void delete()
{
    if(front == -1 || front > rear)
        printf("Queue is Empty");
    else
    {
        printf("Deleted: %d", a[front]);
        front++;

        if(front > rear)
            front = rear = -1;
    }
}

void peek()
{
    if(front == -1)
        printf("Queue is Empty");
    else
        printf("Front value: %d", a[front]);
}

void isempty()
{
    if(front == -1)
        printf("Queue is Empty");
    else
        printf("Queue is not Empty");
}

void isfull()
{
    if(rear == 4)
        printf("Queue is Full");
    else
        printf("Queue is not Full");
}

int main()
{
    int ch;

    while(1)
    {
        printf("\n\n1. Insert");
        printf("\n2. Delete");
        printf("\n3. Peek");
        printf("\n4. IsEmpty");
        printf("\n5. IsFull");
        printf("\n6. Exit");
        printf("\nEnter choice: ");
        scanf("%d", &ch);

        if(ch == 1)
            insert();
        else if(ch == 2)
            delete();
        else if(ch == 3)
            peek();
        else if(ch == 4)
            isempty();
        else if(ch == 5)
            isfull();
        else if(ch == 6)
            break;
        else
            printf("Wrong choice");
    }

    return 0;
}
