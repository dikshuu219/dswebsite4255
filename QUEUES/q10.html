Q10. Implement the operations of double ended queue in an string array (insert, delete, peek, isempty, isfull)

Program

#include <stdio.h>
#include <string.h>

char a[5][50];
int front = -1, rear = -1;

void insertfront()
{
    char x[50];

    if((front == 0 && rear == 4) || (front == rear + 1))
        printf("Queue is Full");
    else
    {
        printf("Enter string: ");
        scanf("%49s", x);

        if(front == -1)
            front = rear = 0;
        else if(front == 0)
            front = 4;
        else
            front--;

        strcpy(a[front], x);
    }
}

void insertrear()
{
    char x[50];

    if((front == 0 && rear == 4) || (front == rear + 1))
        printf("Queue is Full");
    else
    {
        printf("Enter string: ");
        scanf("%49s", x);

        if(front == -1)
            front = rear = 0;
        else if(rear == 4)
            rear = 0;
        else
            rear++;

        strcpy(a[rear], x);
    }
}

void deletefront()
{
    if(front == -1)
        printf("Queue is Empty");
    else
    {
        printf("Deleted: %s", a[front]);

        if(front == rear)
            front = rear = -1;
        else if(front == 4)
            front = 0;
        else
            front++;
    }
}

void deleterear()
{
    if(front == -1)
        printf("Queue is Empty");
    else
    {
        printf("Deleted: %s", a[rear]);

        if(front == rear)
            front = rear = -1;
        else if(rear == 0)
            rear = 4;
        else
            rear--;
    }
}

void peek()
{
    if(front == -1)
        printf("Queue is Empty");
    else
        printf("Front string: %s", a[front]);
}

void isempty()
{
    if(front == -1)
        printf("Queue is Empty");
    else
        printf("Queue is not Empty");
}

void isfull()
{
    if((front == 0 && rear == 4) || (front == rear + 1))
        printf("Queue is Full");
    else
        printf("Queue is not Full");
}

int main()
{
    int ch;

    while(1)
    {
        printf("\n1. Insert Front");
        printf("\n2. Insert Rear");
        printf("\n3. Delete Front");
        printf("\n4. Delete Rear");
        printf("\n5. Peek");
        printf("\n6. IsEmpty");
        printf("\n7. IsFull");
        printf("\n8. Exit");
        printf("\nEnter choice: ");
        scanf("%d", &ch);

        if(ch == 1)
            insertfront();
        else if(ch == 2)
            insertrear();
        else if(ch == 3)
            deletefront();
        else if(ch == 4)
            deleterear();
        else if(ch == 5)
            peek();
        else if(ch == 6)
            isempty();
        else if(ch == 7)
            isfull();
        else if(ch == 8)
            break;
        else
            printf("Wrong choice");
    }

    return 0;
}
