Q20. Rotten Oranges

Program

#include <stdio.h>

int main()
{
    int a[10][10], r, c, i, j, time = 0, fresh = 0, change;

    printf("Enter rows and columns: ");
    scanf("%d %d", &r, &c);

    printf("Enter oranges (0, 1, 2): ");
    for(i = 0; i < r; i++)
        for(j = 0; j < c; j++)
        {
            scanf("%d", &a[i][j]);
            if(a[i][j] == 1)
                fresh++;
        }

    while(fresh > 0)
    {
        change = 0;

        for(i = 0; i < r; i++)
            for(j = 0; j < c; j++)
                if(a[i][j] == 2)
                {
                    if(i > 0 && a[i-1][j] == 1)
                    {
                        a[i-1][j] = 3;
                        fresh--;
                        change = 1;
                    }
                    if(i < r-1 && a[i+1][j] == 1)
                    {
                        a[i+1][j] = 3;
                        fresh--;
                        change = 1;
                    }
                    if(j > 0 && a[i][j-1] == 1)
                    {
                        a[i][j-1] = 3;
                        fresh--;
                        change = 1;
                    }
                    if(j < c-1 && a[i][j+1] == 1)
                    {
                        a[i][j+1] = 3;
                        fresh--;
                        change = 1;
                    }
                }

        if(change == 0)
            break;

        time++;

        for(i = 0; i < r; i++)
            for(j = 0; j < c; j++)
                if(a[i][j] == 3)
                    a[i][j] = 2;
    }

    if(fresh == 0)
        printf("Time: %d minutes", time);
    else
        printf("All oranges cannot rot");

    return 0;
}
