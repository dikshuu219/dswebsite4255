Q19. Queue Reversal

Program

#include <stdio.h>

int main()
{
    int a[100], stack[100];
    int n, i, top = -1;

    printf("Enter number of elements: ");
    scanf("%d", &n);

    printf("Enter queue elements: ");
    for(i = 0; i < n; i++)
        scanf("%d", &a[i]);

    for(i = 0; i < n; i++)
        stack[++top] = a[i];

    for(i = 0; i < n; i++)
        a[i] = stack[top--];

    printf("Reversed Queue: ");
    for(i = 0; i < n; i++)
        printf("%d ", a[i]);

    return 0;
}

Output

Enter number of elements: 5
Enter queue elements: 1
2
3
4
5
Reversed Queue: 5 4 3 2 1
