Q4. Implement the operations of queue dynamically of string (insert, delete, peek, isempty, isfull)

Program

#include <stdio.h>
#include <stdlib.h>
#include <string.h>

struct node
{
    char data[50];
    struct node *next;
};

struct node *front = NULL, *rear = NULL;

void insert()
{
    struct node *p;

    p = (struct node *)malloc(sizeof(struct node));

    if(p == NULL)
        printf("Queue is Full");
    else
    {
        printf("Enter string: ");
        scanf("%49s", p->data);

        p->next = NULL;

        if(front == NULL)
            front = rear = p;
        else
        {
            rear->next = p;
            rear = p;
        }
    }
}

void delete()
{
    struct node *p;

    if(front == NULL)
        printf("Queue is Empty");
    else
    {
        p = front;
        printf("Deleted: %s", front->data);
        front = front->next;

        if(front == NULL)
            rear = NULL;

        free(p);
    }
}

void peek()
{
    if(front == NULL)
        printf("Queue is Empty");
    else
        printf("Front string: %s", front->data);
}

void isempty()
{
    if(front == NULL)
        printf("Queue is Empty");
    else
        printf("Queue is not Empty");
}

void isfull()
{
    struct node *p;

    p = (struct node *)malloc(sizeof(struct node));

    if(p == NULL)
        printf("Queue is Full");
    else
    {
        printf("Queue is not Full");
        free(p);
    }
}

int main()
{
    int ch;

    while(1)
    {
        printf("\n\n1. Insert");
        printf("\n2. Delete");
        printf("\n3. Peek");
        printf("\n4. IsEmpty");
        printf("\n5. IsFull");
        printf("\n6. Exit");
        printf("\nEnter choice: ");
        scanf("%d", &ch);

        if(ch == 1)
            insert();
        else if(ch == 2)
            delete();
        else if(ch == 3)
            peek();
        else if(ch == 4)
            isempty();
        else if(ch == 5)
            isfull();
        else if(ch == 6)
            break;
        else
            printf("Wrong choice");
    }

    return 0;
}
