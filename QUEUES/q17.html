Q17. Queue based approach for first non-repeating character in a stream

Program

#include <stdio.h>

int main()
{
    char s[100], q[100];
    int i, j, front = 0, rear = -1, count;

    printf("Enter string: ");
    scanf("%s", s);

    for(i = 0; s[i] != '\0'; i++)
    {
        q[++rear] = s[i];

        while(front <= rear)
        {
            count = 0;

            for(j = 0; j <= i; j++)
            {
                if(s[j] == q[front])
                    count++;
            }

            if(count == 1)
                break;

            front++;
        }

        if(front <= rear)
            printf("%c ", q[front]);
        else
            printf("-1 ");
    }

    return 0;
}

Output

Enter string: aabc
a -1 b b
