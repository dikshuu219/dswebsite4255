Q14. Length of the longest valid substring

Program

#include <stdio.h>
#include <string.h>

int main()
{
    char s[100];
    int stack[100], top = -1;
    int i, len, max = 0, last = -1;

    printf("Enter brackets: ");
    scanf("%99s", s);

    len = strlen(s);

    for(i = 0; i < len; i++)
    {
        if(s[i] == '(')
            stack[++top] = i;
        else
        {
            if(top == -1)
                last = i;
            else
            {
                top--;

                if(top == -1)
                {
                    if(i - last > max)
                        max = i - last;
                }
                else
                {
                    if(i - stack[top] > max)
                        max = i - stack[top];
                }
            }
        }
    }

    printf("Longest valid substring length: %d", max);

    return 0;
}

Output

Enter brackets: (()())
Longest valid substring length: 6
