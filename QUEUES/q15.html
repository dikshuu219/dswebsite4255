Q15. Find the Next Greater Element

Program

#include <stdio.h>

int main()
{
    int a[100], n, i, j;

    printf("Enter number of elements: ");
    scanf("%d", &n);

    printf("Enter elements: ");
    for(i = 0; i < n; i++)
        scanf("%d", &a[i]);

    for(i = 0; i < n; i++)
    {
        for(j = i + 1; j < n; j++)
        {
            if(a[j] > a[i])
                break;
        }

        if(j < n)
            printf("%d -> %d\n", a[i], a[j]);
        else
            printf("%d -> -1\n", a[i]);
    }

    return 0;
}
