Q8. Implement the operations of priority queue in an string array (insert, delete, peek, isempty, isfull)

Program

#include <stdio.h>
#include <string.h>

char a[5][50];
int front = -1, rear = -1;

void insert()
{
    char x[50];
    int i;

    if(rear == 4)
        printf("Queue is Full");
    else
    {
        printf("Enter string: ");
        scanf("%49s", x);

        if(front == -1)
            front = 0;

        i = rear;

        while(i >= front && strcmp(a[i], x) > 0)
        {
            strcpy(a[i + 1], a[i]);
            i--;
        }

        strcpy(a[i + 1], x);
        rear++;
    }
}

void delete()
{
    if(front == -1)
        printf("Queue is Empty");
    else
    {
        printf("Deleted: %s", a[front]);

        if(front == rear)
            front = rear = -1;
        else
            front++;
    }
}

void peek()
{
    if(front == -1)
        printf("Queue is Empty");
    else
        printf("Highest priority: %s", a[front]);
}

void isempty()
{
    if(front == -1)
        printf("Queue is Empty");
    else
        printf("Queue is not Empty");
}

void isfull()
{
    if(rear == 4)
        printf("Queue is Full");
    else
        printf("Queue is not Full");
}

int main()
{
    int ch;

    while(1)
    {
        printf("\n1. Insert");
        printf("\n2. Delete");
        printf("\n3. Peek");
        printf("\n4. IsEmpty");
        printf("\n5. IsFull");
        printf("\n6. Exit");
        printf("\nEnter choice: ");
        scanf("%d", &ch);

        if(ch == 1)
            insert();
        else if(ch == 2)
            delete();
        else if(ch == 3)
            peek();
        else if(ch == 4)
            isempty();
        else if(ch == 5)
            isfull();
        else if(ch == 6)
            break;
        else
            printf("Wrong choice");
    }

    return 0;
}
