Q11. Enter the 2 polynomial equations dynamically, multiply them and display the answer.

Program

#include <stdio.h>
#include <stdlib.h>

struct node
{
    int coeff;
    int pow;
    struct node *next;
};

struct node *create()
{
    struct node *head = NULL, *p, *q;
    int n, i;

    printf("Enter number of terms: ");
    scanf("%d", &n);

    for (i = 0; i < n; i++)
    {
        p = (struct node *)malloc(sizeof(struct node));

        printf("Enter coefficient and power: ");
        scanf("%d%d", &p->coeff, &p->pow);

        p->next = NULL;

        if (head == NULL)
            head = p;
        else
            q->next = p;

        q = p;
    }

    return head;
}

struct node *insert(struct node *head, int c, int power)
{
    struct node *p, *q;

    if (c == 0)
        return head;

    p = head;

    while (p != NULL)
    {
        if (p->pow == power)
        {
            p->coeff = p->coeff + c;
            return head;
        }

        p = p->next;
    }

    p = (struct node *)malloc(sizeof(struct node));
    p->coeff = c;
    p->pow = power;
    p->next = NULL;

    if (head == NULL)
        return p;

    q = head;

    while (q->next != NULL)
        q = q->next;

    q->next = p;

    return head;
}

struct node *multiply(struct node *p, struct node *q)
{
    struct node *head = NULL, *a, *b;
    int c, power;

    a = p;

    while (a != NULL)
    {
        b = q;

        while (b != NULL)
        {
            c = a->coeff * b->coeff;
            power = a->pow + b->pow;

            head = insert(head, c, power);

            b = b->next;
        }

        a = a->next;
    }

    return head;
}

void display(struct node *p)
{
    int first = 1;

    while (p != NULL)
    {
        if (p->coeff != 0)
        {
            if (!first && p->coeff > 0)
                printf(" + ");

            if (p->coeff < 0)
                printf(" - ");

            if (p->coeff < 0)
                printf("%dx^%d", -p->coeff, p->pow);
            else
                printf("%dx^%d", p->coeff, p->pow);

            first = 0;
        }

        p = p->next;
    }

    if (first)
        printf("0");

    printf("\n");
}

int main()
{
    struct node *p, *q, *result;

    printf("Enter first polynomial:\n");
    p = create();

    printf("Enter second polynomial:\n");
    q = create();

    printf("\nFirst polynomial: ");
    display(p);

    printf("Second polynomial: ");
    display(q);

    result = multiply(p, q);

    printf("Multiplication: ");
    display(result);

    return 0;
}
