Q1. Implement the operations of single linked list storing integer values (insert at begin,insert at pos,insert at end,delete at begin,delete at end, delete at pos, isempty, isfull, display, search)

Program

#include <stdio.h>
#include <stdlib.h>

struct node
{
    int data;
    struct node *next;
};

struct node *head = NULL;

int isempty()
{
    return head == NULL;
}

int isfull()
{
    struct node *p;
    p = (struct node *)malloc(sizeof(struct node));

    if (p == NULL)
        return 1;

    free(p);
    return 0;
}

void insertBegin(int x)
{
    struct node *p;
    p = (struct node *)malloc(sizeof(struct node));

    if (p == NULL)
    {
        printf("List is full\n");
        return;
    }

    p->data = x;
    p->next = head;
    head = p;
}

void insertEnd(int x)
{
    struct node *p, *q;
    p = (struct node *)malloc(sizeof(struct node));

    if (p == NULL)
    {
        printf("List is full\n");
        return;
    }

    p->data = x;
    p->next = NULL;

    if (head == NULL)
    {
        head = p;
        return;
    }

    q = head;
    while (q->next != NULL)
        q = q->next;

    q->next = p;
}

void insertPos(int x, int pos)
{
    struct node *p, *q;
    int i;

    if (pos < 1)
    {
        printf("Invalid position\n");
        return;
    }

    if (pos == 1)
    {
        insertBegin(x);
        return;
    }

    q = head;
    for (i = 1; i < pos - 1 && q != NULL; i++)
        q = q->next;

    if (q == NULL)
    {
        printf("Invalid position\n");
        return;
    }

    p = (struct node *)malloc(sizeof(struct node));

    if (p == NULL)
    {
        printf("List is full\n");
        return;
    }

    p->data = x;
    p->next = q->next;
    q->next = p;
}

void deleteBegin()
{
    struct node *p;

    if (isempty())
    {
        printf("List is empty\n");
        return;
    }

    p = head;
    head = head->next;
    free(p);
}

void deleteEnd()
{
    struct node *p, *q;

    if (isempty())
    {
        printf("List is empty\n");
        return;
    }

    if (head->next == NULL)
    {
        free(head);
        head = NULL;
        return;
    }

    q = head;
    while (q->next->next != NULL)
        q = q->next;

    p = q->next;
    q->next = NULL;
    free(p);
}

void deletePos(int pos)
{
    struct node *p, *q;
    int i;

    if (pos < 1 || isempty())
    {
        printf("Invalid position or empty list\n");
        return;
    }

    if (pos == 1)
    {
        deleteBegin();
        return;
    }

    q = head;
    for (i = 1; i < pos - 1 && q != NULL; i++)
        q = q->next;

    if (q == NULL || q->next == NULL)
    {
        printf("Invalid position\n");
        return;
    }

    p = q->next;
    q->next = p->next;
    free(p);
}

void display()
{
    struct node *p = head;

    if (isempty())
    {
        printf("List is empty\n");
        return;
    }

    while (p != NULL)
    {
        printf("%d -> ", p->data);
        p = p->next;
    }

    printf("NULL\n");
}

void search(int x)
{
    struct node *p = head;
    int pos = 1;

    while (p != NULL)
    {
        if (p->data == x)
        {
            printf("Value found at position %d\n", pos);
            return;
        }

        p = p->next;
        pos++;
    }

    printf("Value not found\n");
}

int main()
{
    int ch, x, pos;

    do
    {
        printf("\n1. Insert at Begin");
        printf("\n2. Insert at Position");
        printf("\n3. Insert at End");
        printf("\n4. Delete at Begin");
        printf("\n5. Delete at End");
        printf("\n6. Delete at Position");
        printf("\n7. Is Empty");
        printf("\n8. Is Full");
        printf("\n9. Display");
        printf("\n10. Search");
        printf("\n0. Exit");
        printf("\nEnter your choice: ");
        scanf("%d", &ch);

        switch (ch)
        {
        case 1:
            printf("Enter value: ");
            scanf("%d", &x);
            insertBegin(x);
            break;

        case 2:
            printf("Enter value and position: ");
            scanf("%d%d", &x, &pos);
            insertPos(x, pos);
            break;

        case 3:
            printf("Enter value: ");
            scanf("%d", &x);
            insertEnd(x);
            break;

        case 4:
            deleteBegin();
            break;

        case 5:
            deleteEnd();
            break;

        case 6:
            printf("Enter position: ");
            scanf("%d", &pos);
            deletePos(pos);
            break;

        case 7:
            if (isempty())
                printf("List is empty\n");
            else
                printf("List is not empty\n");
            break;

        case 8:
            if (isfull())
                printf("List is full\n");
            else
                printf("List is not full\n");
            break;

        case 9:
            display();
            break;

        case 10:
            printf("Enter value to search: ");
            scanf("%d", &x);
            search(x);
            break;

        case 0:
            printf("Program ended\n");
            break;

        default:
            printf("Invalid choice\n");
        }

    } while (ch != 0);

    return 0;
}
