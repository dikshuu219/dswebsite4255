Q8. Implement the operations of doubly circular linked list storing string values (insert at begin,insert at pos,insert at end,delete at begin,delete at end, delete at pos, isempty, isfull, display, search)

Program

#include <stdio.h>
#include <stdlib.h>
#include <string.h>

struct node
{
    char data[50];
    struct node *prev;
    struct node *next;
};

struct node *head = NULL;

int isempty()
{
    return head == NULL;
}

int isfull()
{
    p = (struct node *)malloc(sizeof(struct node));

    if (p == NULL)
        return 1;

    free(p);
    return 0;
}

void insertBegin(char x[])
{
    struct node *p, *last;

    p = (struct node *)malloc(sizeof(struct node));

    if (p == NULL)
        return;

    strcpy(p->data, x);

    if (head == NULL)
    {
        head = p;
        p->next = head;
        p->prev = head;
        return;
    }

    last = head->prev;

    p->next = head;
    p->prev = last;
    last->next = p;
    head->prev = p;
    head = p;
}

void insertEnd(char x[])
{
    struct node *p, *last;

    p = (struct node *)malloc(sizeof(struct node));

    if (p == NULL)
        return;

    strcpy(p->data, x);

    if (head == NULL)
    {
        head = p;
        p->next = head;
        p->prev = head;
        return;
    }

    last = head->prev;

    p->next = head;
    p->prev = last;
    last->next = p;
    head->prev = p;
}

void insertPos(char x[], int pos)
{
    struct node *p, *q;
    int i;

    if (pos < 1)
        return;

    if (pos == 1)
    {
        insertBegin(x);
        return;
    }

    if (head == NULL)
        return;

    q = head;

    for (i = 1; i < pos - 1 && q->next != head; i++)
        q = q->next;

    if (i != pos - 1)
        return;

    p = (struct node *)malloc(sizeof(struct node));

    if (p == NULL)
        return;

    strcpy(p->data, x);

    p->next = q->next;
    p->prev = q;
    q->next->prev = p;
    q->next = p;
}

void deleteBegin()
{
    struct node *p, *last;

    if (head == NULL)
        return;

    p = head;

    if (head->next == head)
    {
        head = NULL;
        free(p);
        return;
    }

    last = head->prev;
    head = head->next;
    head->prev = last;
    last->next = head;

    free(p);
}

void deleteEnd()
{
    struct node *last;

    if (head == NULL)
        return;

    last = head->prev;

    if (last == head)
    {
        head = NULL;
        free(last);
        return;
    }

    last->prev->next = head;
    head->prev = last->prev;

    free(last);
}

void deletePos(int pos)
{
    struct node *p;
    int i;

    if (head == NULL || pos < 1)
        return;

    if (pos == 1)
    {
        deleteBegin();
        return;
    }

    p = head;

    for (i = 1; i < pos && p->next != head; i++)
        p = p->next;

    if (i != pos)
        return;

    p->prev->next = p->next;
    p->next->prev = p->prev;

    free(p);
}

void display()
{
    struct node *p;

    if (head == NULL)
    {
        printf("List is empty\n");
        return;
    }

    p = head;

    do
    {
        printf("%s <-> ", p->data);
        p = p->next;
    } while (p != head);

    printf("HEAD\n");
}

void search(char x[])
{
    struct node *p;
    int pos = 1;

    if (head == NULL)
    {
        printf("List is empty\n");
        return;
    }

    p = head;

    do
    {
        if (strcmp(p->data, x) == 0)
        {
            printf("Found at position %d\n", pos);
            return;
        }

        p = p->next;
        pos++;
    } while (p != head);

    printf("Not found\n");
}

int main()
{
    int ch, pos;
    char x[50];

    do
    {
        printf("\n1. Insert Begin");
        printf("\n2. Insert Position");
        printf("\n3. Insert End");
        printf("\n4. Delete Begin");
        printf("\n5. Delete End");
        printf("\n6. Delete Position");
        printf("\n7. Is Empty");
        printf("\n8. Is Full");
        printf("\n9. Display");
        printf("\n10. Search");
        printf("\n0. Exit");
        printf("\nEnter choice: ");
        scanf("%d", &ch);

        switch (ch)
        {
        case 1:
            printf("Enter string: ");
            scanf("%49s", x);
            insertBegin(x);
            break;

        case 2:
            printf("Enter string and position: ");
            scanf("%49s%d", x, &pos);
            insertPos(x, pos);
            break;

        case 3:
            printf("Enter string: ");
            scanf("%49s", x);
            insertEnd(x);
            break;

        case 4:
            deleteBegin();
            break;

        case 5:
            deleteEnd();
            break;

        case 6:
            printf("Enter position: ");
            scanf("%d", &pos);
            deletePos(pos);
            break;

        case 7:
            if (isempty())
                printf("List is empty\n");
            else
                printf("List is not empty\n");
            break;

        case 8:
            if (isfull())
                printf("List is full\n");
            else
                printf("List is not full\n");
            break;

        case 9:
            display();
            break;

        case 10:
            printf("Enter string to search: ");
            scanf("%49s", x);
            search(x);
            break;

        case 0:
            printf("Exit\n");
            break;

        default:
            printf("Wrong choice\n");
        }

    } while (ch != 0);

    return 0;
}
