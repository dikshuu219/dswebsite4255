Q2. Binary tree level order traversal

Program

#include <stdio.h>
#include <stdlib.h>

struct node
{
    int data;
    struct node *left;
    struct node *right;
};

struct node *create(int value)
{
    struct node *p;

    p = (struct node *)malloc(sizeof(struct node));
    p->data = value;
    p->left = NULL;
    p->right = NULL;

    return p;
}

void levelOrder(struct node *root)
{
    struct node *queue[100];
    int front = 0, rear = 0;

    if (root == NULL)
        return;

    queue[rear++] = root;

    while (front < rear)
    {
        struct node *p = queue[front++];

        printf("%d ", p->data);

        if (p->left != NULL)
            queue[rear++] = p->left;

        if (p->right != NULL)
            queue[rear++] = p->right;
    }
}

int main()
{
    struct node *root;

    root = create(1);
    root->left = create(2);
    root->right = create(3);
    root->left->left = create(4);
    root->left->right = create(5);

    printf("Level Order Traversal: ");
    levelOrder(root);

    return 0;
}

Output

Level Order Traversal: 1 2 3 4 5
