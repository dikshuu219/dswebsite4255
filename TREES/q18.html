Q18. Create almost complete binary tree from the given input by the user (dynamically)

Program

#include <stdio.h>
#include <stdlib.h>

struct node
{
    int data;
    struct node *left, *right;
};

struct node *create()
{
    struct node *p;
    int x;

    printf("Enter value (-1 for no node): ");
    scanf("%d", &x);

    if (x == -1)
        return NULL;

    p = (struct node *)malloc(sizeof(struct node));
    p->data = x;

    printf("Enter left child of %d\n", x);
    p->left = create();

    printf("Enter right child of %d\n", x);
    p->right = create();

    return p;
}

void inorder(struct node *root)
{
    if (root != NULL)
    {
        inorder(root->left);
        printf("%d ", root->data);
        inorder(root->right);
    }
}

int main()
{
    struct node *root;

    printf("Create Almost Complete Binary Tree:\n");
    root = create();

    printf("Inorder Traversal: ");
    inorder(root);

    return 0;
}
