Q6. Find the maximum depth of binary tree

Program

#include <stdio.h>
#include <stdlib.h>

struct node
{
    int data;
    struct node *left, *right;
};

struct node *create()
{
    int x;
    struct node *p;

    printf("Enter data (-1 for no node): ");
    scanf("%d", &x);

    if (x == -1)
        return NULL;

    p = (struct node *)malloc(sizeof(struct node));
    p->data = x;

    printf("Left child of %d:\n", x);
    p->left = create();

    printf("Right child of %d:\n", x);
    p->right = create();

    return p;
}

int depth(struct node *root)
{
    int left, right;

    if (root == NULL)
        return 0;

    left = depth(root->left);
    right = depth(root->right);

    if (left > right)
        return left + 1;
    else
        return right + 1;
}

int main()
{
    struct node *root;

    printf("Create binary tree:\n");
    root = create();

    printf("Maximum depth: %d", depth(root));

    return 0;
}
