Q9. Find inorder predecessor and inorder successor of binary search tree (dynamically represented)

Program

#include <stdio.h>
#include <stdlib.h>

struct node
{
    int data;
    struct node *left, *right;
};

struct node *insert(struct node *root, int x)
{
    struct node *p;

    if (root == NULL)
    {
        p = (struct node *)malloc(sizeof(struct node));
        p->data = x;
        p->left = NULL;
        p->right = NULL;
        return p;
    }

    if (x < root->data)
        root->left = insert(root->left, x);
    else if (x > root->data)
        root->right = insert(root->right, x);

    return root;
}

void findPreSuc(struct node *root, int x)
{
    struct node *pre = NULL;
    struct node *suc = NULL;
    struct node *p = root;

    while (p != NULL)
    {
        if (p->data < x)
        {
            pre = p;
            p = p->right;
        }
        else if (p->data > x)
        {
            suc = p;
            p = p->left;
        }
        else
        {
            if (p->left != NULL)
            {
                pre = p->left;
                while (pre->right != NULL)
                    pre = pre->right;
            }

            if (p->right != NULL)
            {
                suc = p->right;
                while (suc->left != NULL)
                    suc = suc->left;
            }

            break;
        }
    }

    if (pre == NULL)
        printf("Inorder Predecessor not found\n");
    else
        printf("Inorder Predecessor: %d\n", pre->data);

    if (suc == NULL)
        printf("Inorder Successor not found\n");
    else
        printf("Inorder Successor: %d\n", suc->data);
}

int main()
{
    struct node *root = NULL;
    int n, x, i, value;

    printf("Enter number of nodes: ");
    scanf("%d", &n);

    for (i = 0; i < n; i++)
    {
        printf("Enter value: ");
        scanf("%d", &value);
        root = insert(root, value);
    }

    printf("Enter value to find predecessor and successor: ");
    scanf("%d", &x);

    findPreSuc(root, x);

    return 0;
}
