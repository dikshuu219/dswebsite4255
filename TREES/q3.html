Q3. Print left view of binary tree

Program

#include <stdio.h>
#include <stdlib.h>

struct node
{
    int data;
    struct node *left, *right;
};

struct node *create()
{
    int x;
    struct node *p;

    printf("Enter data (-1 for no node): ");
    scanf("%d", &x);

    if (x == -1)
        return NULL;

    p = (struct node *)malloc(sizeof(struct node));
    p->data = x;

    printf("Left child of %d:\n", x);
    p->left = create();

    printf("Right child of %d:\n", x);
    p->right = create();

    return p;
}

void leftView(struct node *root, int level, int *max)
{
    if (root == NULL)
        return;

    if (level > *max)
    {
        printf("%d ", root->data);
        *max = level;
    }

    leftView(root->left, level + 1, max);
    leftView(root->right, level + 1, max);
}

int main()
{
    struct node *root;
    int max = 0;

    printf("Create binary tree:\n");
    root = create();

    printf("Left View: ");
    leftView(root, 1, &max);

    return 0;
}
