Q19. Create complete binary tree from the given input by the user (dynamically)

Program

#include <stdio.h>
#include <stdlib.h>

struct node
{
    int data;
    struct node *left, *right;
};

struct node *createNode(int x)
{
    struct node *p;

    p = (struct node *)malloc(sizeof(struct node));
    p->data = x;
    p->left = NULL;
    p->right = NULL;

    return p;
}

int main()
{
    struct node *a[100];
    struct node *root;
    int n, i, x;

    printf("Enter number of nodes: ");
    scanf("%d", &n);

    if (n <= 0 || n > 100)
    {
        printf("Invalid number of nodes");
        return 0;
    }

    for (i = 0; i < n; i++)
    {
        printf("Enter value: ");
        scanf("%d", &x);
        a[i] = createNode(x);
    }

    for (i = 0; i < n; i++)
    {
        if (2 * i + 1 < n)
            a[i]->left = a[2 * i + 1];

        if (2 * i + 2 < n)
            a[i]->right = a[2 * i + 2];
    }

    root = a[0];

    printf("Complete Binary Tree (Level Order): ");
    for (i = 0; i < n; i++)
        printf("%d ", a[i]->data);

    return 0;
}

Output

Enter value: 1
Enter value: 2
Enter value: 3
Enter value: 4
Enter value: 5
Enter value: 6
Complete Binary Tree (Level Order): 1 2 3 4 5 6
