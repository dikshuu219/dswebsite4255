Q8. Create binary search tree dynamically (insert, delete, search, preorder, inorder, postorder, minimum element)

Program

#include <stdio.h>
#include <stdlib.h>

struct node
{
    int data;
    struct node *left, *right;
};

struct node *insert(struct node *root, int x)
{
    struct node *p;

    if (root == NULL)
    {
        p = (struct node *)malloc(sizeof(struct node));
        p->data = x;
        p->left = NULL;
        p->right = NULL;
        return p;
    }

    if (x < root->data)
        root->left = insert(root->left, x);
    else if (x > root->data)
        root->right = insert(root->right, x);

    return root;
}

void search(struct node *root, int x)
{
    if (root == NULL)
        printf("Not found\n");
    else if (root->data == x)
        printf("Found\n");
    else if (x < root->data)
        search(root->left, x);
    else
        search(root->right, x);
}

struct node *minimum(struct node *root)
{
    while (root != NULL && root->left != NULL)
        root = root->left;

    return root;
}

struct node *deleteNode(struct node *root, int x)
{
    struct node *p;

    if (root == NULL)
        return NULL;

    if (x < root->data)
        root->left = deleteNode(root->left, x);
    else if (x > root->data)
        root->right = deleteNode(root->right, x);
    else
    {
        if (root->left == NULL)
            return root->right;

        if (root->right == NULL)
            return root->left;

        p = minimum(root->right);
        root->data = p->data;
        root->right = deleteNode(root->right, p->data);
    }

    return root;
}

void preorder(struct node *root)
{
    if (root != NULL)
    {
        printf("%d ", root->data);
        preorder(root->left);
        preorder(root->right);
    }
}

void inorder(struct node *root)
{
    if (root != NULL)
    {
        inorder(root->left);
        printf("%d ", root->data);
        inorder(root->right);
    }
}

void postorder(struct node *root)
{
    if (root != NULL)
    {
        postorder(root->left);
        postorder(root->right);
        printf("%d ", root->data);
    }
}

int main()
{
    struct node *root = NULL, *p;
    int ch, x;

    do
    {
        printf("\n1.Insert  2.Delete  3.Search\n");
        printf("4.Preorder  5.Inorder  6.Postorder\n");
        printf("7.Minimum  8.Exit\n");
        printf("Enter choice: ");
        scanf("%d", &ch);

        switch (ch)
        {
            case 1:
                printf("Enter value: ");
                scanf("%d", &x);
                root = insert(root, x);
                break;

            case 2:
                printf("Enter value to delete: ");
                scanf("%d", &x);
                root = deleteNode(root, x);
                break;

            case 3:
                printf("Enter value to search: ");
                scanf("%d", &x);
                search(root, x);
                break;

            case 4:
                preorder(root);
                printf("\n");
                break;

            case 5:
                inorder(root);
                printf("\n");
                break;

            case 6:
                postorder(root);
                printf("\n");
                break;

            case 7:
                p = minimum(root);

                if (p == NULL)
                    printf("Tree is empty\n");
                else
                    printf("Minimum: %d\n", p->data);
                break;

            case 8:
                printf("Exit\n");
                break;

            default:
                printf("Wrong choice\n");
        }

    } while (ch != 8);

    return 0;
}
