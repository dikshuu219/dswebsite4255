Q1. Construct binary tree from preorder and inorder traversal

Program

#include <stdio.h>
#include <stdlib.h>

struct node
{
    char data;
    struct node *left;
    struct node *right;
};

int index = 0;

struct node *create(char value)
{
    struct node *p;

    p = (struct node *)malloc(sizeof(struct node));

    p->data = value;
    p->left = NULL;
    p->right = NULL;

    return p;
}

int find(char inorder[], char value, int n)
{
    int i;

    for (i = 0; i < n; i++)
    {
        if (inorder[i] == value)
            return i;
    }

    return -1;
}

struct node *build(char preorder[], char inorder[], int n)
{
    struct node *root;
    int pos;

    if (n <= 0)
        return NULL;

    root = create(preorder[index]);
    index++;

    pos = find(inorder, root->data, n);

    root->left = build(preorder, inorder, pos);
    root->right = build(preorder + index, inorder + pos + 1, n - pos - 1);

    return root;
}

void display(struct node *root)
{
    if (root != NULL)
    {
        display(root->left);
        printf("%c ", root->data);
        display(root->right);
    }
}

int main()
{
    char preorder[20], inorder[20];
    struct node *root;
    int n;

    printf("Enter number of nodes: ");
    scanf("%d", &n);

    printf("Enter preorder: ");
    scanf("%19s", preorder);

    printf("Enter inorder: ");
    scanf("%19s", inorder);

    index = 0;
    root = build(preorder, inorder, n);

    printf("Constructed tree inorder: ");
    display(root);

    return 0;
}

Output

Enter number of nodes: 3
Enter preorder: ABC
Enter inorder: CBA
Constructed tree inorder: C B A
