Q5. Construct binary tree from postorder and inorder traversal

Program

#include <stdio.h>
#include <stdlib.h>

struct node
{
    char data;
    struct node *left, *right;
};

int index;

struct node *create(char x)
{
    struct node *p;

    p = (struct node *)malloc(sizeof(struct node));
    p->data = x;
    p->left = NULL;
    p->right = NULL;

    return p;
}

int find(char inorder[], char x, int start, int end)
{
    int i;

    for (i = start; i <= end; i++)
    {
        if (inorder[i] == x)
            return i;
    }

    return -1;
}

struct node *build(char postorder[], char inorder[],
                   int start, int end)
{
    struct node *root;
    int pos;

    if (start > end)
        return NULL;

    root = create(postorder[index]);
    index--;

    if (start == end)
        return root;

    pos = find(inorder, root->data, start, end);

    root->right = build(postorder, inorder, pos + 1, end);
    root->left = build(postorder, inorder, start, pos - 1);

    return root;
}

void display(struct node *root)
{
    if (root != NULL)
    {
        display(root->left);
        display(root->right);
        printf("%c ", root->data);
    }
}

int main()
{
    char postorder[50], inorder[50];
    struct node *root;
    int n;

    printf("Enter number of nodes: ");
    scanf("%d", &n);

    printf("Enter postorder: ");
    scanf("%49s", postorder);

    printf("Enter inorder: ");
    scanf("%49s", inorder);

    index = n - 1;

    root = build(postorder, inorder, 0, n - 1);

    printf("Postorder of constructed tree: ");
    display(root);

    return 0;
}
