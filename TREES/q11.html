Q11. Create binary tree to binary search tree through array.

Program

#include <stdio.h>
#include <stdlib.h>

struct node
{
    int data;
    struct node *left, *right;
};

struct node *createTree()
{
    struct node *p;
    int x;

    printf("Enter value (-1 for no node): ");
    scanf("%d", &x);

    if (x == -1)
        return NULL;

    p = (struct node *)malloc(sizeof(struct node));
    p->data = x;

    printf("Enter left child of %d\n", x);
    p->left = createTree();

    printf("Enter right child of %d\n", x);
    p->right = createTree();

    return p;
}

void store(struct node *root, int a[], int *n)
{
    if (root != NULL)
    {
        store(root->left, a, n);
        a[*n] = root->data;
        (*n)++;
        store(root->right, a, n);
    }
}

void sort(int a[], int n)
{
    int i, j, temp;

    for (i = 0; i < n - 1; i++)
    {
        for (j = 0; j < n - i - 1; j++)
        {
            if (a[j] > a[j + 1])
            {
                temp = a[j];
                a[j] = a[j + 1];
                a[j + 1] = temp;
            }
        }
    }
}

void putValues(struct node *root, int a[], int *i)
{
    if (root != NULL)
    {
        putValues(root->left, a, i);
        root->data = a[*i];
        (*i)++;
        putValues(root->right, a, i);
    }
}

void inorder(struct node *root)
{
    if (root != NULL)
    {
        inorder(root->left);
        printf("%d ", root->data);
        inorder(root->right);
    }
}

int main()
{
    struct node *root;
    int a[100], n = 0, i = 0;

    printf("Create Binary Tree:\n");
    root = createTree();

    store(root, a, &n);
    sort(a, n);
    putValues(root, a, &i);

    printf("\nBinary Search Tree (Inorder): ");
    inorder(root);

    return 0;
}
