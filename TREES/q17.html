Q17. Create strictly binary tree from the given input by the user (dynamically)

Program

#include <stdio.h>
#include <stdlib.h>

struct node
{
    int data;
    struct node *left, *right;
};

struct node *create()
{
    struct node *p;
    int x;

    printf("Enter value (-1 for no node): ");
    scanf("%d", &x);

    if (x == -1)
        return NULL;

    p = (struct node *)malloc(sizeof(struct node));
    p->data = x;

    printf("Does %d have children? (1 for Yes, 0 for No): ", x);
    scanf("%d", &x);

    if (x == 1)
    {
        printf("Enter left child of %d\n", p->data);
        p->left = create();

        printf("Enter right child of %d\n", p->data);
        p->right = create();
    }
    else
    {
        p->left = NULL;
        p->right = NULL;
    }

    return p;
}

void preorder(struct node *root)
{
    if (root != NULL)
    {
        printf("%d ", root->data);
        preorder(root->left);
        preorder(root->right);
    }
}

int main()
{
    struct node *root;

    printf("Create Strictly Binary Tree:\n");
    root = create();

    printf("Preorder traversal: ");
    preorder(root);

    return 0;
}

Enter right child of 1
Enter value (-1 for no node): 3
