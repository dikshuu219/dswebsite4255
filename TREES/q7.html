Q7. Create binary search tree using arrays (insert, delete, search, preorder, inorder, postorder, minimum element)

Program

#include <stdio.h>

int a[100];

void insert(int x)
{
    int i = 1;

    while (i < 100)
    {
        if (a[i] == -1)
        {
            a[i] = x;
            break;
        }
        else if (x < a[i])
            i = 2 * i;
        else if (x > a[i])
            i = 2 * i + 1;
        else
            break;
    }
}

void search(int i, int x)
{
    if (i >= 100 || a[i] == -1)
        printf("Element not found\n");
    else if (a[i] == x)
        printf("Element found\n");
    else if (x < a[i])
        search(2 * i, x);
    else
        search(2 * i + 1, x);
}

void inorder(int i)
{
    if (i < 100 && a[i] != -1)
    {
        inorder(2 * i);
        printf("%d ", a[i]);
        inorder(2 * i + 1);
    }
}

void preorder(int i)
{
    if (i < 100 && a[i] != -1)
    {
        printf("%d ", a[i]);
        preorder(2 * i);
        preorder(2 * i + 1);
    }
}

void postorder(int i)
{
    if (i < 100 && a[i] != -1)
    {
        postorder(2 * i);
        postorder(2 * i + 1);
        printf("%d ", a[i]);
    }
}

void deleteNode(int x)
{
    int i = 1, j, k;

    while (i < 100 && a[i] != -1 && a[i] != x)
    {
        if (x < a[i])
            i = 2 * i;
        else
            i = 2 * i + 1;
    }

    if (i >= 100 || a[i] == -1)
    {
        printf("Element not found\n");
        return;
    }

    if (i * 2 < 100 && a[i * 2] != -1)
    {
        j = i * 2;

        while (j * 2 + 1 < 100 && a[j * 2 + 1] != -1)
            j = j * 2 + 1;
    }
    else if (i * 2 + 1 < 100 && a[i * 2 + 1] != -1)
    {
        j = i * 2 + 1;

        while (j * 2 < 100 && a[j * 2] != -1)
            j = j * 2;
    }
    else
    {
        a[i] = -1;
        return;
    }

    a[i] = a[j];
    a[j] = -1;

    k = j;

    while (k > 1)
    {
        if (a[k] != -1)
            break;
        k = k / 2;
    }
}

int main()
{
    int i, ch, x;

    for (i = 0; i < 100; i++)
        a[i] = -1;

    do
    {
        printf("\n1.Insert  2.Delete  3.Search\n");
        printf("4.Preorder  5.Inorder  6.Postorder\n");
        printf("7.Minimum  8.Exit\n");
        printf("Enter choice: ");
        scanf("%d", &ch);

        switch (ch)
        {
            case 1:
                printf("Enter value: ");
                scanf("%d", &x);
                insert(x);
                break;

            case 2:
                printf("Enter value to delete: ");
                scanf("%d", &x);
                deleteNode(x);
                break;

            case 3:
                printf("Enter value to search: ");
                scanf("%d", &x);
                search(1, x);
                break;

            case 4:
                preorder(1);
                printf("\n");
                break;

            case 5:
                inorder(1);
                printf("\n");
                break;

            case 6:
                postorder(1);
                printf("\n");
                break;

            case 7:
                i = 1;

                if (a[i] == -1)
                    printf("Tree is empty\n");
                else
                {
                    while (2 * i < 100 && a[2 * i] != -1)
                        i = 2 * i;

                    printf("Minimum element: %d\n", a[i]);
                }
                break;

            case 8:
                printf("Exit\n");
                break;

            default:
                printf("Wrong choice\n");
        }

    } while (ch != 8);

    return 0;
}
