Q12. Find kth largest element in binary search tree through array

Program

#include <stdio.h>

int main()
{
    int a[100], n, i, j, temp, k;

    printf("Enter number of elements: ");
    scanf("%d", &n);

    printf("Enter elements: ");
    for (i = 0; i < n; i++)
        scanf("%d", &a[i]);

    for (i = 0; i < n - 1; i++)
    {
        for (j = i + 1; j < n; j++)
        {
            if (a[i] < a[j])
            {
                temp = a[i];
                a[i] = a[j];
                a[j] = temp;
            }
        }
    }

    printf("Enter K: ");
    scanf("%d", &k);

    if (k >= 1 && k <= n)
        printf("Kth largest element: %d\n", a[k - 1]);
    else
        printf("Invalid K\n");

    return 0;
}
