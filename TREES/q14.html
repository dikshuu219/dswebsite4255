Q14. Print preorder traversal to postorder traversal of binary search tree

Program

#include <stdio.h>

int main()
{
    int a[100], n, i;

    printf("Enter number of nodes: ");
    scanf("%d", &n);

    printf("Enter preorder elements: ");
    for (i = 0; i < n; i++)
        scanf("%d", &a[i]);

    printf("Postorder traversal: ");

    for (i = n - 1; i >= 0; i--)
        printf("%d ", a[i]);

    return 0;
}

Output

Enter number of nodes: 7
Enter preorder elements: 50
30
20
40
70
60
80
Postorder traversal: 80 60 70 40 20 30 50
