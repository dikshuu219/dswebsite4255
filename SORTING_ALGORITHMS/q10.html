Q10. Enter a integer array from the user and perform quick sort algorithm through recursion. also calculate the worst and best complexity with exact time.

Program

#include <stdio.h>

int comparisons = 0;

void quickSort(int a[], int low, int high)
{
    int i, j, pivot, temp;

    if(low < high)
    {
        pivot = a[high];
        i = low - 1;

        for(j = low; j < high; j++)
        {
            comparisons++;

            if(a[j] < pivot)
            {
                i++;

                temp = a[i];
                a[i] = a[j];
                a[j] = temp;
            }
        }

        temp = a[i + 1];
        a[i + 1] = a[high];
        a[high] = temp;

        quickSort(a, low, i);
        quickSort(a, i + 2, high);
    }
}

int main()
{
    int a[100], size, i;

    printf("Enter number of elements: ");
    scanf("%d", &size);

    printf("Enter array elements:\n");
    for(i = 0; i < size; i++)
        scanf("%d", &a[i]);

    quickSort(a, 0, size - 1);

    printf("Sorted array:\n");
    for(i = 0; i < size; i++)
        printf("%d ", a[i]);

    printf("\nTotal comparisons = %d", comparisons);
    printf("\nBest Case Time Complexity = O(n log n)");
    printf("\nWorst Case Time Complexity = O(n^2)");

    return 0;
}
