Q15. Enter a string array from the user and perform insertion sort algorithm through iteration. also calculate the worst and best complexity with exact time.

Program

#include <stdio.h>
#include <string.h>

int main()
{
    char a[100][50], key[50];
    int size, i, j, comparisons = 0;

    printf("Enter number of strings: ");
    scanf("%d", &size);

    printf("Enter strings:\n");
    for(i = 0; i < size; i++)
        scanf("%49s", a[i]);

    for(i = 1; i < size; i++)
    {
        strcpy(key, a[i]);
        j = i - 1;

        while(j >= 0)
        {
            comparisons++;

            if(strcmp(a[j], key) > 0)
            {
                strcpy(a[j + 1], a[j]);
                j--;
            }
            else
                break;
        }

        strcpy(a[j + 1], key);
    }

    printf("Sorted strings:\n");
    for(i = 0; i < size; i++)
        printf("%s ", a[i]);

    printf("\nTotal comparisons = %d", comparisons);
    printf("\nBest Case Time Complexity = O(n)");
    printf("\nWorst Case Time Complexity = O(n^2)");

    return 0;
}

Output

Enter strings:

apple

Sorted strings:

Total comparisons = 4
Best Case Time Complexity = O(n)
Worst Case Time Complexity = O(n^2)
