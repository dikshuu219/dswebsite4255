Q2. Enter a integer array from the user and perform selection sort algorithm through recursion. also calculate the worst and best complexity with exact time.

Program

#include <stdio.h>

int comparisons = 0;

void selectionSort(int a[], int size, int i)
{
    int j, minimum, temp;

    if (i >= size - 1)
        return;

    minimum = i;

    for (j = i + 1; j < size; j++)
    {
        comparisons++;

        if (a[j] < a[minimum])
            minimum = j;
    }

    temp = a[i];
    a[i] = a[minimum];
    a[minimum] = temp;

    selectionSort(a, size, i + 1);
}

int main()
{
    int a[100], size, i;

    printf("Enter number of elements: ");
    scanf("%d", &size);

    printf("Enter array elements:\n");
    for (i = 0; i < size; i++)
        scanf("%d", &a[i]);

    selectionSort(a, size, 0);

    printf("Sorted array in ascending order:\n");
    for (i = 0; i < size; i++)
        printf("%d ", a[i]);

    printf("\nTotal comparisons = %d", comparisons);
    printf("\nBest Case Time Complexity = O(n^2)");
    printf("\nWorst Case Time Complexity = O(n^2)");

    return 0;
}
