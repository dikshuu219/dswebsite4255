Q7. Enter a integer array from the user and perform merge sort algorithm through iteration. also calculate the worst and best complexity with exact time.

Program

#include <stdio.h>

int main()
{
    int a[100], temp[100], size, i, j, k;
    int left, mid, right, width;
    int comparisons = 0;

    printf("Enter number of elements: ");
    scanf("%d", &size);

    printf("Enter array elements:\n");
    for(i = 0; i < size; i++)
        scanf("%d", &a[i]);

    for(width = 1; width < size; width = width * 2)
    {
        for(left = 0; left < size; left = left + 2 * width)
        {
            mid = left + width - 1;
            right = left + 2 * width - 1;

            if(mid >= size)
                continue;

            if(right >= size)
                right = size - 1;

            i = left;
            j = mid + 1;
            k = left;

            while(i <= mid && j <= right)
            {
                comparisons++;

                if(a[i] <= a[j])
                    temp[k++] = a[i++];
                else
                    temp[k++] = a[j++];
            }

            while(i <= mid)
                temp[k++] = a[i++];

            while(j <= right)
                temp[k++] = a[j++];

            for(i = left; i <= right; i++)
                a[i] = temp[i];
        }
    }

    printf("Sorted array:\n");
    for(i = 0; i < size; i++)
        printf("%d ", a[i]);

    printf("\nTotal comparisons = %d", comparisons);
    printf("\nBest Case Time Complexity = O(n log n)");
    printf("\nWorst Case Time Complexity = O(n log n)");

    return 0;
}
