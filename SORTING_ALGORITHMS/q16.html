Q16. Enter a string array from the user and perform insertion sort algorithm through recursion. also calculate the worst and best complexity with exact time.

Program

#include <stdio.h>
#include <string.h>

int comparisons = 0;

void insertionSort(char a[][50], int size)
{
    int j;
    char key[50];

    if(size <= 1)
        return;

    insertionSort(a, size - 1);

    strcpy(key, a[size - 1]);
    j = size - 2;

    while(j >= 0)
    {
        comparisons++;

        if(strcmp(a[j], key) > 0)
        {
            strcpy(a[j + 1], a[j]);
            j--;
        }
        else
            break;
    }

    strcpy(a[j + 1], key);
}

int main()
{
    char a[100][50];
    int size, i;

    printf("Enter number of strings: ");
    scanf("%d", &size);

    printf("Enter strings:\n");
    for(i = 0; i < size; i++)
        scanf("%49s", a[i]);

    insertionSort(a, size);

    printf("Sorted strings:\n");
    for(i = 0; i < size; i++)
        printf("%s ", a[i]);

    printf("\nTotal comparisons = %d", comparisons);
    printf("\nBest Case Time Complexity = O(n)");
    printf("\nWorst Case Time Complexity = O(n^2)");

    return 0;
}
