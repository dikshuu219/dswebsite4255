Q8. Enter a integer array from the user and perform merge sort algorithm through recursion. also calculate the worst and best complexity with exact time.

Program

#include <stdio.h>

int comparisons = 0;

void merge(int a[], int left, int mid, int right)
{
    int temp[100];
    int i = left, j = mid + 1, k = left;

    while(i <= mid && j <= right)
    {
        comparisons++;

        if(a[i] <= a[j])
            temp[k++] = a[i++];
        else
            temp[k++] = a[j++];
    }

    while(i <= mid)
        temp[k++] = a[i++];

    while(j <= right)
        temp[k++] = a[j++];

    for(i = left; i <= right; i++)
        a[i] = temp[i];
}

void mergeSort(int a[], int left, int right)
{
    int mid;

    if(left < right)
    {
        mid = (left + right) / 2;

        mergeSort(a, left, mid);
        mergeSort(a, mid + 1, right);

        merge(a, left, mid, right);
    }
}

int main()
{
    int a[100], size, i;

    printf("Enter number of elements: ");
    scanf("%d", &size);

    printf("Enter array elements:\n");
    for(i = 0; i < size; i++)
        scanf("%d", &a[i]);

    mergeSort(a, 0, size - 1);

    printf("Sorted array:\n");
    for(i = 0; i < size; i++)
        printf("%d ", a[i]);

    printf("\nTotal comparisons = %d", comparisons);
    printf("\nBest Case Time Complexity = O(n log n)");
    printf("\nWorst Case Time Complexity = O(n log n)");

    return 0;
}
