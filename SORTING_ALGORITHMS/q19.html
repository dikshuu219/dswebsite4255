Q19. Enter a string array from the user and perform quick sort algorithm through iteration. also calculate the worst and best complexity with exact time.

Program

#include <stdio.h>
#include <string.h>

int main()
{
    char a[100][50], temp[50];
    int stack[100];
    int size, i, j, low, high, top;
    int pivot, comparisons = 0;

    printf("Enter number of strings: ");
    scanf("%d", &size);

    printf("Enter strings:\n");
    for(i = 0; i < size; i++)
        scanf("%49s", a[i]);

    top = -1;
    stack[++top] = 0;
    stack[++top] = size - 1;

    while(top >= 0)
    {
        high = stack[top--];
        low = stack[top--];

        pivot = high;
        i = low - 1;

        for(j = low; j < high; j++)
        {
            comparisons++;

            if(strcmp(a[j], a[pivot]) < 0)
            {
                i++;

                strcpy(temp, a[i]);
                strcpy(a[i], a[j]);
                strcpy(a[j], temp);
            }
        }

        strcpy(temp, a[i + 1]);
        strcpy(a[i + 1], a[high]);
        strcpy(a[high], temp);

        pivot = i + 1;

        if(pivot - 1 > low)
        {
            stack[++top] = low;
            stack[++top] = pivot - 1;
        }

        if(pivot + 1 < high)
        {
            stack[++top] = pivot + 1;
            stack[++top] = high;
        }
    }

    printf("\nSorted strings:\n");
    for(i = 0; i < size; i++)
        printf("%s ", a[i]);

    printf("\nTotal comparisons = %d", comparisons);
    printf("\nBest Case Time Complexity = O(n log n)");
    printf("\nWorst Case Time Complexity = O(n^2)");

    return 0;
}

Output

Worst Case Time Complexity = O(n^2)
