Q9. Enter a integer array from the user and perform quick sort algorithm through iteration. also calculate the worst and best complexity with exact time.

Program

#include <stdio.h>

int main()
{
    int a[100], stack[100], size, i, j;
    int low, high, pivot, temp, top = -1;
    int comparisons = 0;

    printf("Enter number of elements: ");
    scanf("%d", &size);

    printf("Enter array elements:\n");
    for(i = 0; i < size; i++)
        scanf("%d", &a[i]);

    if(size > 1)
    {
        stack[++top] = 0;
        stack[++top] = size - 1;
    }

    while(top >= 0)
    {
        high = stack[top--];
        low = stack[top--];

        pivot = a[high];
        i = low - 1;

        for(j = low; j < high; j++)
        {
            comparisons++;

            if(a[j] < pivot)
            {
                i++;
                temp = a[i];
                a[i] = a[j];
                a[j] = temp;
            }
        }

        temp = a[i + 1];
        a[i + 1] = a[high];
        a[high] = temp;

        j = i + 1;

        if(j - 1 > low)
        {
            stack[++top] = low;
            stack[++top] = j - 1;
        }

        if(j + 1 < high)
        {
            stack[++top] = j + 1;
            stack[++top] = high;
        }
    }

    printf("Sorted array:\n");
    for(i = 0; i < size; i++)
        printf("%d ", a[i]);

    printf("\nTotal comparisons = %d", comparisons);
    printf("\nBest Case Time Complexity = O(n log n)");
    printf("\nWorst Case Time Complexity = O(n^2)");

    return 0;
}
