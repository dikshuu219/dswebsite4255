Q17. Enter a string array from the user and perform merge sort algorithm through iteration. also calculate the worst and best complexity with exact time.

Program

#include <stdio.h>
#include <string.h>

int main()
{
    char a[100][50], temp[100][50];
    int size, i, j, k, left, mid, right, width;
    int comparisons = 0;

    printf("Enter number of strings: ");
    scanf("%d", &size);

    printf("Enter strings:\n");
    for(i = 0; i < size; i++)
        scanf("%49s", a[i]);

    for(width = 1; width < size; width = width * 2)
    {
        for(left = 0; left < size; left = left + 2 * width)
        {
            mid = left + width;
            right = left + 2 * width;

            if(mid > size)
                mid = size;

            if(right > size)
                right = size;

            i = left;
            j = mid;
            k = left;

            while(i < mid && j < right)
            {
                comparisons++;

                if(strcmp(a[i], a[j]) <= 0)
                    strcpy(temp[k++], a[i++]);
                else
                    strcpy(temp[k++], a[j++]);
            }

            while(i < mid)
                strcpy(temp[k++], a[i++]);

            while(j < right)
                strcpy(temp[k++], a[j++]);

            for(i = left; i < right; i++)
                strcpy(a[i], temp[i]);
        }
    }

    printf("Sorted strings:\n");
    for(i = 0; i < size; i++)
        printf("%s ", a[i]);

    printf("\nTotal comparisons = %d", comparisons);
    printf("\nBest Case Time Complexity = O(n log n)");
    printf("\nWorst Case Time Complexity = O(n log n)");

    return 0;
}
