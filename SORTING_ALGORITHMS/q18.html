Q18. Enter a string array from the user and perform merge sort algorithm through recursion. also calculate the worst and best complexity with exact time

Program

#include <stdio.h>
#include <string.h>

int comparisons = 0;

void merge(char a[][50], int left, int mid, int right)
{
    char temp[100][50];
    int i = left, j = mid + 1, k = left;

    while(i <= mid && j <= right)
    {
        comparisons++;

        if(strcmp(a[i], a[j]) <= 0)
            strcpy(temp[k++], a[i++]);
        else
            strcpy(temp[k++], a[j++]);
    }

    while(i <= mid)
        strcpy(temp[k++], a[i++]);

    while(j <= right)
        strcpy(temp[k++], a[j++]);

    for(i = left; i <= right; i++)
        strcpy(a[i], temp[i]);
}

void mergeSort(char a[][50], int left, int right)
{
    int mid;

    if(left < right)
    {
        mid = (left + right) / 2;

        mergeSort(a, left, mid);
        mergeSort(a, mid + 1, right);

        merge(a, left, mid, right);
    }
}

int main()
{
    char a[100][50];
    int size, i;

    printf("Enter number of strings: ");
    scanf("%d", &size);

    printf("Enter strings:\n");
    for(i = 0; i < size; i++)
        scanf("%49s", a[i]);

    mergeSort(a, 0, size - 1);

    printf("Sorted strings:\n");
    for(i = 0; i < size; i++)
        printf("%s ", a[i]);

    printf("\nTotal comparisons = %d", comparisons);
    printf("\nBest Case Time Complexity = O(n log n)");
    printf("\nWorst Case Time Complexity = O(n log n)");

    return 0;
}
