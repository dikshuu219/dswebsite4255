Q6. Enter a integer array from the user and perform insertion sort algorithm through recursion. also calculate the worst and best complexity with exact time.

Program

#include <stdio.h>

int comparisons = 0;

void insertionSort(int a[], int size)
{
    int key, j;

    if (size <= 1)
        return;

    insertionSort(a, size - 1);

    key = a[size - 1];
    j = size - 2;

    while (j >= 0 && a[j] > key)
    {
        comparisons++;
        a[j + 1] = a[j];
        j--;
    }

    if (j >= 0)
        comparisons++;

    a[j + 1] = key;
}

int main()
{
    int a[100], size, i;

    printf("Enter number of elements: ");
    scanf("%d", &size);

    printf("Enter array elements:\n");
    for (i = 0; i < size; i++)
        scanf("%d", &a[i]);

    // Insertion sort using recursion
    insertionSort(a, size);

    printf("Sorted array in ascending order:\n");
    for (i = 0; i < size; i++)
        printf("%d ", a[i]);

    printf("\nTotal comparisons = %d", comparisons);
    printf("\nBest Case Time Complexity = O(n)");
    printf("\nWorst Case Time Complexity = O(n^2)");

    return 0;
}
