Q12. Enter a string array from the user and perform selection sort algorithm through recursion. also calculate the worst and best complexity with exact time.

Program

#include <stdio.h>
#include <string.h>

int comparisons = 0;

void selectionSort(char a[][50], int size, int index)
{
    int i, min;
    char temp[50];

    if(index == size - 1)
        return;

    min = index;

    for(i = index + 1; i < size; i++)
    {
        comparisons++;

        if(strcmp(a[i], a[min]) < 0)
            min = i;
    }

    if(min != index)
    {
        strcpy(temp, a[index]);
        strcpy(a[index], a[min]);
        strcpy(a[min], temp);
    }

    selectionSort(a, size, index + 1);
}

int main()
{
    char a[100][50];
    int size, i;

    printf("Enter number of strings: ");
    scanf("%d", &size);

    printf("Enter strings:\n");
    for(i = 0; i < size; i++)
        scanf("%49s", a[i]);

    selectionSort(a, size, 0);

    printf("Sorted strings:\n");
    for(i = 0; i < size; i++)
        printf("%s ", a[i]);

    printf("\nTotal comparisons = %d", comparisons);
    printf("\nBest Case Time Complexity = O(n^2)");
    printf("\nWorst Case Time Complexity = O(n^2)");

    return 0;
}
