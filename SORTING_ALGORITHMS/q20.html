Q20. Enter a string array from the user and perform quick sort algorithm through recursion. also calculate the worst and best complexity with exact time.

Program

#include <stdio.h>
#include <string.h>

int comparisons = 0;

void quickSort(char a[][50], int low, int high)
{
    int i, j, pivot;
    char temp[50];

    if(low < high)
    {
        pivot = high;
        i = low - 1;

        for(j = low; j < high; j++)
        {
            comparisons++;

            if(strcmp(a[j], a[pivot]) < 0)
            {
                i++;

                strcpy(temp, a[i]);
                strcpy(a[i], a[j]);
                strcpy(a[j], temp);
            }
        }

        strcpy(temp, a[i + 1]);
        strcpy(a[i + 1], a[high]);
        strcpy(a[high], temp);

        pivot = i + 1;

        quickSort(a, low, pivot - 1);
        quickSort(a, pivot + 1, high);
    }
}

int main()
{
    char a[100][50];
    int size, i;

    printf("Enter number of strings: ");
    scanf("%d", &size);

    printf("Enter strings:\n");
    for(i = 0; i < size; i++)
        scanf("%49s", a[i]);

    quickSort(a, 0, size - 1);

    printf("\nSorted strings:\n");
    for(i = 0; i < size; i++)
        printf("%s ", a[i]);

    printf("\nTotal comparisons = %d", comparisons);
    printf("\nBest Case Time Complexity = O(n log n)");
    printf("\nWorst Case Time Complexity = O(n^2)");

    return 0;
}
